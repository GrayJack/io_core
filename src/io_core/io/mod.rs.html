65// Several `read_to_string` and `read_line` methods in the standard library will
66// append data into a `String` buffer, but we need to be pretty careful when
67// doing this. The implementation will just call `.as_mut_vec()` and then
68// delegate to a byte-oriented reading method, but we must ensure that when
69// returning we never leave `buf` in a state such that it contains invalid UTF-8
70// in its bounds.
71//
72// To this end, we use an RAII guard (to protect against panics) which updates
73// the length of the string when it is dropped. This guard initially truncates
74// the string to the prior length and only after we've validated that the
75// new contents are valid UTF-8 do we allow it to set a longer length.
76//
77// The unsafety in this function is twofold:
78//
79// 1. We're looking at the raw bytes of `buf`, so we take on the burden of UTF-8 checks.
80// 2. We're passing a raw buffer to the function `f`, and it is expected that the function only
81//    *appends* bytes to the buffer. We'll get undefined behavior if existing bytes are overwritten
82//    to have non-UTF-8 data.
83#[cfg(feature = "alloc")]
84pub(crate) unsafe fn append_to_string<F>(buf: &mut String, f: F) -> Result<usize>
85where
86    F: FnOnce(&mut Vec<u8>) -> Result<usize>,
87{
88    let mut g = Guard {
89        len: buf.len(),
90        buf: unsafe { buf.as_mut_vec() },
91    };
92    let ret = f(g.buf);
93
94    // SAFETY: the caller promises to only append data to `buf`
95    let appended = unsafe { g.buf.get_unchecked(g.len..) };
96    if str::from_utf8(appended).is_err() {
97        ret.and_then(|_| Err(Error::INVALID_UTF8))
98    } else {
99        g.len = g.buf.len();
100        ret
101    }
102}

104// Here we must serve many masters with conflicting goals:
105//
106// - avoid allocating unless necessary
107// - avoid overallocating if we know the exact size (#89165)
108// - avoid passing large buffers to readers that always initialize the free capacity if they perform
109//   short reads (#23815, #23820)
110// - pass large buffers to readers that do not initialize the spare capacity. this can amortize
111//   per-call overheads
112// - and finally pass not-too-small and not-too-large buffers to Windows read APIs because they
113//   manage to suffer from both problems at the same time, i.e. small reads suffer from syscall
114//   overhead, all reads incur costs proportional to buffer size (#110650)
115//
116#[cfg(feature = "alloc")]
117pub(crate) fn default_read_to_end<R: Read + ?Sized>(
118    r: &mut R, buf: &mut Vec<u8>, size_hint: Option<usize>,
119) -> Result<usize> {
120    let start_len = buf.len();
121    let start_cap = buf.capacity();
122    // Optionally limit the maximum bytes read on each iteration.
123    // This adds an arbitrary fiddle factor to allow for more data than we expect.
124    let mut max_read_size = size_hint
125        .and_then(|s| s.checked_add(1024)?.checked_next_multiple_of(DEFAULT_BUF_SIZE))
126        .unwrap_or(DEFAULT_BUF_SIZE);
127
128    const PROBE_SIZE: usize = 32;
129
130    fn small_probe_read<R: Read + ?Sized>(r: &mut R, buf: &mut Vec<u8>) -> Result<usize> {
131        let mut probe = [0u8; PROBE_SIZE];
132
133        loop {
134            match r.read(&mut probe) {
135                Ok(n) => {
136                    // there is no way to recover from allocation failure here
137                    // because the data has already been read.
138                    buf.extend_from_slice(&probe[..n]);
139                    return Ok(n);
140                },
141                Err(ref e) if e.is_interrupted() => continue,
142                Err(e) => return Err(e),
143            }
144        }
145    }
146
147    // avoid inflating empty/small vecs before we have determined that there's anything to read
148    if (size_hint.is_none() || size_hint == Some(0)) && buf.capacity() - buf.len() < PROBE_SIZE {
149        let read = small_probe_read(r, buf)?;
150
151        if read == 0 {
152            return Ok(0);
153        }
154    }
155
156    loop {
157        if buf.len() == buf.capacity() && buf.capacity() == start_cap {
158            // The buffer might be an exact fit. Let's read into a probe buffer
159            // and see if it returns `Ok(0)`. If so, we've avoided an
160            // unnecessary doubling of the capacity. But if not, append the
161            // probe buffer to the primary buffer and let its capacity grow.
162            let read = small_probe_read(r, buf)?;
163
164            if read == 0 {
165                return Ok(buf.len() - start_len);
166            }
167        }
168
169        if buf.len() == buf.capacity() {
170            // buf is full, need more space
171            buf.try_reserve(PROBE_SIZE)?;
172        }
173
174        let mut spare = buf.spare_capacity_mut();
175        let buf_len = cmp::min(spare.len(), max_read_size);
176        spare = &mut spare[..buf_len];
177        let mut read_buf: BorrowedBuf<'_, u8> = spare.into();
178
179        // Note that we don't track already initialized bytes here, but this is fine
180        // because we explicitly limit the read size
181        let mut cursor = read_buf.unfilled();
182        let result = loop {
183            match r.read_buf(cursor.reborrow()) {
184                Err(e) if e.is_interrupted() => continue,
185                // Do not stop now in case of error: we might have received both data
186                // and an error
187                res => break res,
188            }
189        };
190
191        let bytes_read = cursor.written();
192        let is_init = read_buf.is_init();
193
194        // SAFETY: BorrowedBuf's invariants mean this much memory is initialized.
195        unsafe {
196            let new_len = bytes_read + buf.len();
197            buf.set_len(new_len);
198        }
199
200        // Now that all data is pushed to the vector, we can fail without data loss
201        result?;
202
203        if bytes_read == 0 {
204            return Ok(buf.len() - start_len);
205        }
206
207        // Use heuristics to determine the max read size if no initial size hint was provided
208        if size_hint.is_none() {
209            // The reader is returning short reads but it doesn't call ensure_init().
210            // In that case we no longer need to restrict read sizes to avoid
211            // initialization costs.
212            // When reading from disk we usually don't get any short reads except at EOF.
213            // So we wait for at least 2 short reads before uncapping the read buffer;
214            // this helps with the Windows issue.
215            if !is_init {
216                max_read_size = usize::MAX;
217            }
218            // we have passed a larger buffer than previously and the
219            // reader still hasn't returned a short read
220            else if buf_len >= max_read_size && bytes_read == buf_len {
221                max_read_size = max_read_size.saturating_mul(2);
222            }
223        }
224    }
225}

227#[cfg(feature = "alloc")]
228pub(crate) fn default_read_to_string<R: Read + ?Sized>(
229    r: &mut R, buf: &mut String, size_hint: Option<usize>,
230) -> Result<usize> {
231    // Note that we do *not* call `r.read_to_end()` here. We are passing
232    // `&mut Vec<u8>` (the raw contents of `buf`) into the `read_to_end`
233    // method to fill it up. An arbitrary implementation could overwrite the
234    // entire contents of the vector, not just append to it (which is what
235    // we are expecting).
236    //
237    // To prevent extraneously checking the UTF-8-ness of the entire buffer
238    // we pass it to our hardcoded `default_read_to_end` implementation which
239    // we know is guaranteed to only read data into the end of the buffer.
240    unsafe { append_to_string(buf, |b| default_read_to_end(r, b, size_hint)) }
241}

259pub(crate) fn default_read_exact<R: Read + ?Sized>(this: &mut R, mut buf: &mut [u8]) -> Result<()> {
260    while !buf.is_empty() {
261        match this.read(buf) {
262            Ok(0) => break,
263            Ok(n) => {
264                buf = &mut buf[n..];
265            },
266            Err(ref e) if e.is_interrupted() => {},
267            Err(e) => return Err(e),
268        }
269    }
270    if !buf.is_empty() {
271        Err(Error::READ_EXACT_EOF)
272    } else {
273        Ok(())
274    }
275}

286pub(crate) fn default_read_buf_exact<R: Read + ?Sized>(
287    this: &mut R, mut cursor: BorrowedCursor<'_, u8>,
288) -> Result<()> {
289    while cursor.capacity() > 0 {
290        let prev_written = cursor.written();
291        match this.read_buf(cursor.reborrow()) {
292            Ok(()) => {},
293            Err(e) if e.is_interrupted() => continue,
294            Err(e) => return Err(e),
295        }
296
297        if cursor.written() == prev_written {
298            return Err(Error::READ_EXACT_EOF);
299        }
300    }
301
302    Ok(())
303}

305pub(crate) fn default_write_fmt<W: Write + ?Sized>(
306    this: &mut W, args: fmt::Arguments<'_>,
307) -> Result<()> {
308    // Create a shim which translates a `Write` to a `fmt::Write` and saves off
309    // I/O errors, instead of discarding them.
310    struct Adapter<'a, T: ?Sized + 'a> {
311        inner: &'a mut T,
312        error: Result<()>,
313    }
314
315    impl<T: Write + ?Sized> fmt::Write for Adapter<'_, T> {
316        fn write_str(&mut self, s: &str) -> fmt::Result {
317            match self.inner.write_all(s.as_bytes()) {
318                Ok(()) => Ok(()),
319                Err(e) => {
320                    self.error = Err(e);
321                    Err(fmt::Error)
322                },
323            }
324        }
325    }
326
327    let mut output = Adapter {
328        inner: this,
329        error: Ok(()),
330    };
331    match fmt::write(&mut output, args) {
332        Ok(()) => Ok(()),
333        Err(..) => {
334            // Check whether the error came from the underlying `Write`.
335            if output.error.is_err() {
336                output.error
337            } else {
338                // This shouldn't happen: the underlying stream did not error,
339                // but somehow the formatter still errored?
340                panic!(
341                    "a formatting trait implementation returned an error when the underlying \
342                     stream did not"
343                );
344            }
345        },
346    }
347}

349/// The `Read` trait allows for reading bytes from a source.
350///
351/// Implementors of the `Read` trait are called 'readers'.
352///
353/// Readers are defined by one required method, [`read()`]. Each call to [`read()`]
354/// will attempt to pull bytes from this source into a provided buffer. A
355/// number of other methods are implemented in terms of [`read()`], giving
356/// implementors a number of ways to read bytes while only needing to implement
357/// a single method.
358///
359/// Readers are intended to be composable with one another. Many implementors
360/// throughout [`std::io`] take and provide types which implement the `Read`
361/// trait.
362///
363/// Please note that each call to [`read()`] may involve a system call, and
364/// therefore, using something that implements [`BufRead`], such as
365/// [`BufReader`], will be more efficient.
366///
367/// Repeated calls to the reader use the same cursor, so for example
368/// calling `read_to_end` twice on a [`File`] will only return the file's
369/// contents once. It's recommended to first call `rewind()` in that case.
370///
371/// # Examples
372///
373/// [`File`]s implement `Read`:
374///
375/// ```no_run
376/// use std::fs::File;
377///
378/// use io_core::io::{self, Read};
379///
380/// fn main() -> io::Result<()> {
381///     let mut f = File::open("foo.txt")?;
382///     let mut buffer = [0; 10];
383///
384///     // read up to 10 bytes
385///     f.read(&mut buffer)?;
386///
387///     let mut buffer = Vec::new();
388///     // read the whole file
389///     f.read_to_end(&mut buffer)?;
390///
391///     // read into a String, so that you don't need to do the conversion.
392///     let mut buffer = String::new();
393///     f.read_to_string(&mut buffer)?;
394///
395///     // and more! See the other methods for more details.
396///     Ok(())
397/// }
398/// ```
399///
400/// Read from [`&str`] because [`&[u8]`][prim@slice] implements `Read`:
401///
402/// ```no_run
403/// # use io_core::io;
404/// use io_core::io::Read;
405///
406/// fn main() -> io::Result<()> {
407///     let mut b = "This string will be read".as_bytes();
408///     let mut buffer = [0; 10];
409///
410///     // read up to 10 bytes
411///     b.read(&mut buffer)?;
412///
413///     // etc... it works exactly as a File does!
414///     Ok(())
415/// }
416/// ```
417///
418/// [`read()`]: Read::read
419/// [`&str`]: prim@str
420/// [`std::io`]: self
421/// [`File`]: std::fs::File
422pub trait Read {
423    /// Pull some bytes from this source into the specified buffer, returning
424    /// how many bytes were read.
425    ///
426    /// This function does not provide any guarantees about whether it blocks
427    /// waiting for data, but if an object needs to block for a read and cannot,
428    /// it will typically signal this via an [`Err`] return value.
429    ///
430    /// If the return value of this method is [`Ok(n)`], then implementations must
431    /// guarantee that `0 <= n <= buf.len()`. A nonzero `n` value indicates
432    /// that the buffer `buf` has been filled in with `n` bytes of data from this
433    /// source. If `n` is `0`, then it can indicate one of two scenarios:
434    ///
435    /// 1. This reader has reached its "end of file" and will likely no longer be able to produce
436    ///    bytes. Note that this does not mean that the reader will *always* no longer be able to
437    ///    produce bytes. As an example, on Linux, this method will call the `recv` syscall for a
438    ///    [`TcpStream`], where returning zero indicates the connection was shut down correctly.
439    ///    While for [`File`], it is possible to reach the end of file and get zero as result, but
440    ///    if more data is appended to the file, future calls to `read` will return more data.
441    /// 2. The buffer specified was 0 bytes in length.
442    ///
443    /// It is not an error if the returned value `n` is smaller than the buffer size,
444    /// even when the reader is not at the end of the stream yet.
445    /// This may happen for example because fewer bytes are actually available right now
446    /// (e. g. being close to end-of-file) or because read() was interrupted by a signal.
447    ///
448    /// As this trait is safe to implement, callers in unsafe code cannot rely on
449    /// `n <= buf.len()` for safety.
450    /// Extra care needs to be taken when `unsafe` functions are used to access the read bytes.
451    /// Callers have to ensure that no unchecked out-of-bounds accesses are possible even if
452    /// `n > buf.len()`.
453    ///
454    /// *Implementations* of this method can make no assumptions about the contents of `buf` when
455    /// this function is called. It is recommended that implementations only write data to `buf`
456    /// instead of reading its contents.
457    ///
458    /// Correspondingly, however, *callers* of this method in unsafe code must not assume
459    /// any guarantees about how the implementation uses `buf`. The trait is safe to implement,
460    /// so it is possible that the code that's supposed to write to the buffer might also read
461    /// from it. It is your responsibility to make sure that `buf` is initialized
462    /// before calling `read`. Calling `read` with an uninitialized `buf` (of the kind one
463    /// obtains via [`MaybeUninit<T>`]) is not safe, and can lead to undefined behavior.
464    ///
465    /// [`MaybeUninit<T>`]: core::mem::MaybeUninit
466    ///
467    /// # Errors
468    ///
469    /// If this function encounters any form of I/O or other error, an error
470    /// variant will be returned. If an error is returned then it must be
471    /// guaranteed that no bytes were read.
472    ///
473    /// An error of the [`ErrorKind::Interrupted`] kind is non-fatal and the read
474    /// operation should be retried if there is nothing else to do.
475    ///
476    /// # Examples
477    ///
478    /// [`File`]s implement `Read`:
479    ///
480    /// [`Ok(n)`]: Ok
481    /// [`File`]: std::fs::File
482    /// [`TcpStream`]: std::net::TcpStream
483    ///
484    /// ```no_run
485    /// use std::fs::File;
486    ///
487    /// use io_core::io::{self, Read};
488    ///
489    /// fn main() -> io::Result<()> {
490    ///     let mut f = File::open("foo.txt")?;
491    ///     let mut buffer = [0; 10];
492    ///
493    ///     // read up to 10 bytes
494    ///     let n = f.read(&mut buffer[..])?;
495    ///
496    ///     println!("The bytes: {:?}", &buffer[..n]);
497    ///     Ok(())
498    /// }
499    /// ```
500    fn read(&mut self, buf: &mut [u8]) -> Result<usize>;
501
502    /// Like `read`, except that it reads into a slice of buffers.
503    ///
504    /// Data is copied to fill each buffer in order, with the final buffer
505    /// written to possibly being only partially filled. This method must
506    /// behave equivalently to a single call to `read` with concatenated
507    /// buffers.
508    ///
509    /// The default implementation calls `read` with either the first nonempty
510    /// buffer provided, or an empty one if none exists.
511    fn read_vectored(&mut self, bufs: &mut [IoSliceMut<'_>]) -> Result<usize> {
512        default_read_vectored(|b| self.read(b), bufs)
513    }
514
515    /// Determines if this `Read`er has an efficient `read_vectored`
516    /// implementation.
517    ///
518    /// If a `Read`er does not override the default `read_vectored`
519    /// implementation, code using it may want to avoid the method all together
520    /// and coalesce writes into a single buffer for higher performance.
521    ///
522    /// The default implementation returns `false`.
523    fn is_read_vectored(&self) -> bool {
524        false
525    }
526
527    /// Reads all bytes until EOF in this source, placing them into `buf`.
528    ///
529    /// All bytes read from this source will be appended to the specified buffer
530    /// `buf`. This function will continuously call [`read()`] to append more data to
531    /// `buf` until [`read()`] returns either [`Ok(0)`] or an error of
532    /// non-[`ErrorKind::Interrupted`] kind.
533    ///
534    /// If successful, this function will return the total number of bytes read.
535    ///
536    /// # Errors
537    ///
538    /// If this function encounters an error of the kind
539    /// [`ErrorKind::Interrupted`] then the error is ignored and the operation
540    /// will continue.
541    ///
542    /// If any other read error is encountered then this function immediately
543    /// returns. Any bytes which have already been read will be appended to
544    /// `buf`.
545    ///
546    /// # Examples
547    ///
548    /// [`File`]s implement `Read`:
549    ///
550    /// [`read()`]: Read::read
551    /// [`Ok(0)`]: Ok
552    /// [`File`]: std::fs::File
553    ///
554    /// ```no_run
555    /// use std::fs::File;
556    ///
557    /// use io_core::io::{self, Read};
558    ///
559    /// fn main() -> io::Result<()> {
560    ///     let mut f = File::open("foo.txt")?;
561    ///     let mut buffer = Vec::new();
562    ///
563    ///     // read the whole file
564    ///     f.read_to_end(&mut buffer)?;
565    ///     Ok(())
566    /// }
567    /// ```
568    ///
569    /// (See also the [`std::fs::read`] convenience function for reading from a
570    /// file.)
571    ///
572    /// [`std::fs::read`]: std::fs::read
573    ///
574    /// ## Implementing `read_to_end`
575    ///
576    /// When implementing the `io::Read` trait, it is recommended to allocate
577    /// memory using [`Vec::try_reserve`]. However, this behavior is not guaranteed
578    /// by all implementations, and `read_to_end` may not handle out-of-memory
579    /// situations gracefully.
580    ///
581    /// ```no_run
582    /// # use io_core::io::{self, BufRead};
583    /// # struct Example { example_datasource: io::Empty } impl Example {
584    /// # fn get_some_data_for_the_example(&self) -> &'static [u8] { &[] }
585    /// fn read_to_end(&mut self, dest_vec: &mut Vec<u8>) -> io::Result<usize> {
586    ///     let initial_vec_len = dest_vec.len();
587    ///     loop {
588    ///         let src_buf = self.example_datasource.fill_buf()?;
589    ///         if src_buf.is_empty() {
590    ///             break;
591    ///         }
592    ///         dest_vec.try_reserve(src_buf.len())?;
593    ///         dest_vec.extend_from_slice(src_buf);
594    ///
595    ///         // Any irreversible side effects should happen after `try_reserve` succeeds,
596    ///         // to avoid losing data on allocation error.
597    ///         let read = src_buf.len();
598    ///         self.example_datasource.consume(read);
599    ///     }
600    ///     Ok(dest_vec.len() - initial_vec_len)
601    /// }
602    /// # }
603    /// ```
604    ///
605    /// # Usage Notes
606    ///
607    /// `read_to_end` attempts to read a source until EOF, but many sources are continuous streams
608    /// that do not send EOF. In these cases, `read_to_end` will block indefinitely. Standard input
609    /// is one such stream which may be finite if piped, but is typically continuous. For example,
610    /// `cat file | my-rust-program` will correctly terminate with an `EOF` upon closure of cat.
611    /// Reading user input or running programs that remain open indefinitely will never terminate
612    /// the stream with `EOF` (e.g. `yes | my-rust-program`).
613    ///
614    /// Using `.lines()` with a [`BufReader`] or using [`read`] can provide a better solution
615    ///
616    /// [`read`]: Read::read
617    ///
618    /// [`Vec::try_reserve`]: alloc::vec::Vec::try_reserve
619    #[cfg(feature = "alloc")]
620    fn read_to_end(&mut self, buf: &mut Vec<u8>) -> Result<usize> {
621        default_read_to_end(self, buf, None)
622    }
623
624    /// Reads all bytes until EOF in this source, appending them to `buf`.
625    ///
626    /// If successful, this function returns the number of bytes which were read
627    /// and appended to `buf`.
628    ///
629    /// # Errors
630    ///
631    /// If the data in this stream is *not* valid UTF-8 then an error is

952    ///     assert_eq!(z, 0x504);

958    where
959        Self: Sized,

962        let mut borrowed_buf = BorrowedBuf::from(buf.as_mut_slice());

964        // Guard against incorrect `read_buf_exact` implementations.
965        assert_eq!(borrowed_buf.len(), N);
966        Ok(unsafe { MaybeUninit::array_assume_init(buf) })
967    }

970/// Reads all bytes from a [reader][Read] into a new [`String`].
971///
972/// This is a convenience function for [`Read::read_to_string`]. Using this
973/// function avoids having to create a variable first and provides more type
974/// safety since you can only get the buffer out if there were no errors. (If you
975/// use [`Read::read_to_string`] you have to remember to check whether the read
976/// succeeded because otherwise your buffer will be empty or only partially full.)
977///
978/// # Performance
979///
980/// The downside of this function's increased ease of use and type safety is
981/// that it gives you less control over performance. For example, you can't
982/// pre-allocate memory like you can using [`String::with_capacity`] and
983/// [`Read::read_to_string`]. Also, you can't re-use the buffer if an error
984/// occurs while reading.
985///
986/// In many cases, this function's performance will be adequate and the ease of use
987/// and type safety tradeoffs will be worth it. However, there are cases where you
988/// need more control over performance, and in those cases you should definitely use
989/// [`Read::read_to_string`] directly.
990///
991/// Note that in some special cases, such as when reading files, this function will
992/// pre-allocate memory based on the size of the input it is reading. In those
993/// cases, the performance should be as good as if you had used
994/// [`Read::read_to_string`] with a manually pre-allocated buffer.
995///
996/// # Errors
997///
998/// This function forces you to handle errors because the output (the `String`)
999/// is wrapped in a [`Result`]. See [`Read::read_to_string`] for the errors
1000/// that can occur. If any error occurs, you will get an [`Err`], so you
1001/// don't have to worry about your buffer being empty or partially full.
1002///
1003/// # Examples
1004///
1005/// ```no_run
1006/// # use io_core::io;
1007/// fn main() -> io::Result<()> {
1008///     let stdin = io::read_to_string(std::io::stdin())?;
1009///     println!("Stdin was:");
1010///     println!("{stdin}");
1011///     Ok(())
1012/// }
1013/// ```
1014///
1015/// # Usage Notes
1016///
1017/// `read_to_string` attempts to read a source until EOF, but many sources are continuous streams
1018/// that do not send EOF. In these cases, `read_to_string` will block indefinitely. Standard input
1019/// is one such stream which may be finite if piped, but is typically continuous. For example,
1020/// `cat file | my-rust-program` will correctly terminate with an `EOF` upon closure of cat.
1021/// Reading user input or running programs that remain open indefinitely will never terminate
1022/// the stream with `EOF` (e.g. `yes | my-rust-program`).
1023///
1024/// Using `.lines()` with a [`BufReader`] or using [`read`] can provide a better solution
1025///
1026/// [`read`]: Read::read
1027#[cfg(feature = "alloc")]
1028pub fn read_to_string<R: Read>(mut reader: R) -> Result<String> {
1029    let mut buf = String::new();
1030    reader.read_to_string(&mut buf)?;
1031    Ok(buf)
1032}

1271    /// IoSlice::advance_slices(&mut bufs, 10);
1272    /// assert_eq!(bufs[0].deref(), [2; 14].as_ref());
1273    /// assert_eq!(bufs[1].deref(), [3; 8].as_ref());
1274    #[inline]
1275    pub fn advance_slices(bufs: &mut &mut [IoSlice<'a>], n: usize) {
1276        // Number of buffers to remove.
1277        let mut remove = 0;
1278        // Remaining length before reaching n. This prevents overflow
1279        // that could happen if the length of slices in `bufs` were instead
1280        // accumulated. Those slice may be aliased and, if they are large
1281        // enough, their added length may overflow a `usize`.
1282        let mut left = n;
1283        for buf in bufs.iter() {
1284            if let Some(remainder) = left.checked_sub(buf.len()) {
1285                left = remainder;
1286                remove += 1;
1287            } else {
1288                break;
1289            }
1290        }
1291
1292        *bufs = &mut mem::take(bufs)[remove..];
1293        if bufs.is_empty() {
1294            assert!(left == 0, "advancing io slices beyond their length");
1295        } else {
1296            bufs[0].advance(left);
1297        }
1298    }
1299
1300    /// Get the underlying bytes as a slice with the original lifetime.
1301    ///
1302    /// This doesn't borrow from `self`, so is less restrictive than calling
1303    /// `.deref()`, which does.
1304    ///
1305    /// # Examples
1306    ///
1307    /// ```
1308    /// use io_core::io::IoSlice;
1309    ///
1310    /// let data = b"abcdef";
1311    ///
1312    /// let mut io_slice = IoSlice::new(data);
1313    /// let tail = &io_slice.as_slice()[3..];
1314    ///
1315    /// // This works because `tail` doesn't borrow `io_slice`
1316    /// io_slice = IoSlice::new(tail);
1317    ///
1318    /// assert_eq!(io_slice.as_slice(), b"def");
1319    /// ```
1320    pub const fn as_slice(self) -> &'a [u8] {
1321        self.0.as_slice()
1322    }
1323}

1334/// A trait for objects which are byte-oriented sinks.
1335///
1336/// Implementors of the `Write` trait are sometimes called 'writers'.
1337///
1338/// Writers are defined by two required methods, [`write`] and [`flush`]:
1339///
1340/// * The [`write`] method will attempt to write some data into the object, returning how many bytes
1341///   were successfully written.
1342///
1343/// * The [`flush`] method is useful for adapters and explicit buffers themselves for ensuring that
1344///   all buffered data has been pushed out to the 'true sink'.
1345///
1346/// Writers are intended to be composable with one another. Many implementors
1347/// throughout [`std::io`] take and provide types which implement the `Write`
1348/// trait.
1349///
1350/// [`write`]: Write::write
1351/// [`flush`]: Write::flush
1352/// [`std::io`]: self
1353///
1354/// # Examples
1355///
1356/// ```no_run
1357/// use std::fs::File;
1358///
1359/// use io_core::io::{self, Write};
1360///
1361/// fn main() -> io::Result<()> {
1362///     let data = b"some bytes";
1363///
1364///     let mut pos = 0;
1365///     let mut buffer = File::create("foo.txt")?;
1366///
1367///     while pos < data.len() {
1368///         let bytes_written = buffer.write(&data[pos..])?;
1369///         pos += bytes_written;
1370///     }
1371///     Ok(())
1372/// }
1373/// ```
1374///
1375/// The trait also provides convenience methods like [`write_all`], which calls
1376/// `write` in a loop until its entire input has been written.
1377///
1378/// [`write_all`]: Write::write_all
1379pub trait Write {
1380    /// Writes a buffer into this writer, returning how many bytes were written.
1381    ///
1382    /// This function will attempt to write the entire contents of `buf`, but
1383    /// the entire write might not succeed, or the write may also generate an
1384    /// error. Typically, a call to `write` represents one attempt to write to
1385    /// any wrapped object.
1386    ///
1387    /// Calls to `write` are not guaranteed to block waiting for data to be
1388    /// written, and a write which would otherwise block can be indicated through
1389    /// an [`Err`] variant.
1390    ///
1391    /// If this method consumed `n > 0` bytes of `buf` it must return [`Ok(n)`].
1392    /// If the return value is `Ok(n)` then `n` must satisfy `n <= buf.len()`.
1393    /// A return value of `Ok(0)` typically means that the underlying object is
1394    /// no longer able to accept bytes and will likely not be able to in the
1395    /// future as well, or that the buffer provided is empty.
1396    ///
1397    /// # Errors
1398    ///
1399    /// Each call to `write` may generate an I/O error indicating that the
1400    /// operation could not be completed. If an error is returned then no bytes
1401    /// in the buffer were written to this writer.
1402    ///
1403    /// It is **not** considered an error if the entire buffer could not be
1404    /// written to this writer.
1405    ///
1406    /// An error of the [`ErrorKind::Interrupted`] kind is non-fatal and the
1407    /// write operation should be retried if there is nothing else to do.
1408    ///
1409    /// # Examples
1410    ///
1411    /// ```no_run
1412    /// use std::fs::File;
1413    ///
1414    /// use io_core::io::{self, Write};
1415    ///
1416    /// fn main() -> io::Result<()> {
1417    ///     let mut buffer = File::create("foo.txt")?;
1418    ///
1419    ///     // Writes some prefix of the byte string, not necessarily all of it.
1420    ///     buffer.write(b"some bytes")?;
1421    ///     Ok(())
1422    /// }
1423    /// ```
1424    ///
1425    /// [`Ok(n)`]: Ok
1426    fn write(&mut self, buf: &[u8]) -> Result<usize>;
1427
1428    /// Like [`write`], except that it writes from a slice of buffers.
1429    ///
1430    /// Data is copied from each buffer in order, with the final buffer
1431    /// read from possibly being only partially consumed. This method must
1432    /// behave as a call to [`write`] with the buffers concatenated would.
1433    ///
1434    /// The default implementation calls [`write`] with either the first nonempty
1435    /// buffer provided, or an empty one if none exists.
1436    ///
1437    /// # Examples
1438    ///
1439    /// ```no_run
1440    /// use std::fs::File;
1441    ///
1442    /// use io_core::io::{self, IoSlice, Write};
1443    ///
1444    /// fn main() -> io::Result<()> {
1445    ///     let data1 = [1; 8];
1446    ///     let data2 = [15; 8];
1447    ///     let io_slice1 = IoSlice::new(&data1);
1448    ///     let io_slice2 = IoSlice::new(&data2);
1449    ///
1450    ///     let mut buffer = File::create("foo.txt")?;
1451    ///
1452    ///     // Writes some prefix of the byte string, not necessarily all of it.
1453    ///     buffer.write_vectored(&[io_slice1, io_slice2])?;
1454    ///     Ok(())
1455    /// }
1456    /// ```
1457    ///
1458    /// [`write`]: Write::write
1459    fn write_vectored(&mut self, bufs: &[IoSlice<'_>]) -> Result<usize> {
1460        default_write_vectored(|b| self.write(b), bufs)
1461    }
1462
1463    /// Determines if this `Write`r has an efficient [`write_vectored`]
1464    /// implementation.
1465    ///
1466    /// If a `Write`r does not override the default [`write_vectored`]
1467    /// implementation, code using it may want to avoid the method all together
1468    /// and coalesce writes into a single buffer for higher performance.
1469    ///
1470    /// The default implementation returns `false`.
1471    ///
1472    /// [`write_vectored`]: Write::write_vectored
1473    fn is_write_vectored(&self) -> bool {
1474        false
1475    }
1476
1477    /// Flushes this output stream, ensuring that all intermediately buffered
1478    /// contents reach their destination.
1479    ///
1480    /// # Errors
1481    ///
1482    /// It is considered an error if not all bytes could be written due to
1483    /// I/O errors or EOF being reached.
1484    ///
1485    /// # Examples
1486    ///
1487    /// ```no_run
1488    /// use std::fs::File;
1489    ///
1490    /// use io_core::io::{self, BufWriter, Write};
1491    ///
1492    /// fn main() -> io::Result<()> {
1493    ///     let mut buffer = BufWriter::new(File::create("foo.txt")?);
1494    ///
1495    ///     buffer.write_all(b"some bytes")?;
1496    ///     buffer.flush()?;
1497    ///     Ok(())
1498    /// }
1499    /// ```
1500    fn flush(&mut self) -> Result<()>;
1501
1502    /// Attempts to write an entire buffer into this writer.
1503    ///
1504    /// This method will continuously call [`write`] until there is no more data
1505    /// to be written or an error of non-[`ErrorKind::Interrupted`] kind is
1506    /// returned. This method will not return until the entire buffer has been
1507    /// successfully written or such an error occurs. The first error that is
1508    /// not of [`ErrorKind::Interrupted`] kind generated from this method will be
1509    /// returned.
1510    ///
1511    /// If the buffer contains no data, this will never call [`write`].
1512    ///
1513    /// # Errors
1514    ///
1515    /// This function will return the first error of
1516    /// non-[`ErrorKind::Interrupted`] kind that [`write`] returns.
1517    ///
1518    /// [`write`]: Write::write
1519    ///
1520    /// # Examples
1521    ///
1522    /// ```no_run
1523    /// use std::fs::File;
1524    ///
1525    /// use io_core::io::{self, Write};
1526    ///
1527    /// fn main() -> io::Result<()> {
1528    ///     let mut buffer = File::create("foo.txt")?;
1529    ///
1530    ///     buffer.write_all(b"some bytes")?;
1531    ///     Ok(())
1532    /// }
1533    /// ```
1534    fn write_all(&mut self, mut buf: &[u8]) -> Result<()> {
1535        while !buf.is_empty() {
1536            match self.write(buf) {
1537                Ok(0) => {
1538                    return Err(Error::WRITE_ALL_EOF);
1539                },
1540                Ok(n) => buf = &buf[n..],
1541                Err(ref e) if e.is_interrupted() => {},
1542                Err(e) => return Err(e),
1543            }
1544        }
1545        Ok(())
1546    }
1547
1548    /// Attempts to write multiple buffers into this writer.
1549    ///
1550    /// This method will continuously call [`write_vectored`] until there is no
1551    /// more data to be written or an error of non-[`ErrorKind::Interrupted`]
1552    /// kind is returned. This method will not return until all buffers have
1553    /// been successfully written or such an error occurs. The first error that
1554    /// is not of [`ErrorKind::Interrupted`] kind generated from this method
1555    /// will be returned.
1556    ///
1557    /// If the buffer contains no data, this will never call [`write_vectored`].
1558    ///
1559    /// # Notes
1560    ///
1561    /// Unlike [`write_vectored`], this takes a *mutable* reference to
1562    /// a slice of [`IoSlice`]s, not an immutable one. That's because we need to
1563    /// modify the slice to keep track of the bytes already written.
1564    ///
1565    /// Once this function returns, the contents of `bufs` are unspecified, as
1566    /// this depends on how many calls to [`write_vectored`] were necessary. It is
1567    /// best to understand this function as taking ownership of `bufs` and to
1568    /// not use `bufs` afterwards. The underlying buffers, to which the
1569    /// [`IoSlice`]s point (but not the [`IoSlice`]s themselves), are unchanged and
1570    /// can be reused.
1571    ///
1572    /// [`write_vectored`]: Write::write_vectored
1573    ///
1574    /// # Examples
1575    ///
1576    /// ```
1577    /// # fn main() -> io_core::io::Result<()> {
1578    ///
1579    /// use io_core::io::{IoSlice, Write};
1580    ///
1581    /// let mut writer = Vec::new();
1582    /// let bufs = &mut [IoSlice::new(&[1]), IoSlice::new(&[2, 3]), IoSlice::new(&[4, 5, 6])];
1583    ///
1584    /// writer.write_all_vectored(bufs)?;
1585    /// // Note: the contents of `bufs` is now undefined, see the Notes section.
1586    ///
1587    /// assert_eq!(writer, &[1, 2, 3, 4, 5, 6]);
1588    /// # Ok(()) }
1589    /// ```
1590    fn write_all_vectored(&mut self, mut bufs: &mut [IoSlice<'_>]) -> Result<()> {
1591        // Guarantee that bufs is empty if it contains no data,
1592        // to avoid calling write_vectored if there is no data to be written.
1593        IoSlice::advance_slices(&mut bufs, 0);
1594        while !bufs.is_empty() {
1595            match self.write_vectored(bufs) {
1596                Ok(0) => {
1597                    return Err(Error::WRITE_ALL_EOF);
1598                },
1599                Ok(n) => IoSlice::advance_slices(&mut bufs, n),
1600                Err(ref e) if e.is_interrupted() => {},
1601                Err(e) => return Err(e),
1602            }
1603        }
1604        Ok(())
1605    }
1606
1607    /// Writes a formatted string into this writer, returning any error
1608    /// encountered.
1609    ///
1610    /// This method is primarily used to interface with the
1611    /// [`format_args!()`] macro, and it is rare that this should
1612    /// explicitly be called. The [`write!()`] macro should be favored to
1613    /// invoke this method instead.
1614    ///
1615    /// This function internally uses the [`write_all`] method on
1616    /// this trait and hence will continuously write data so long as no errors
1617    /// are received. This also means that partial writes are not indicated in
1618    /// this signature.
1619    ///
1620    /// [`write_all`]: Write::write_all
1621    ///
1622    /// # Errors
1623    ///
1624    /// This function will return any I/O error reported while formatting.
1625    ///
1626    /// # Examples
1627    ///
1628    /// ```no_run
1629    /// use std::fs::File;
1630    ///
1631    /// use io_core::io::{self, Write};
1632    ///
1633    /// fn main() -> io::Result<()> {
1634    ///     let mut buffer = File::create("foo.txt")?;
1635    ///
1636    ///     // this call
1637    ///     write!(buffer, "{:.*}", 2, 1.234567)?;
1638    ///     // turns into this:
1639    ///     buffer.write_fmt(format_args!("{:.*}", 2, 1.234567))?;
1640    ///     Ok(())
1641    /// }
1642    /// ```
1643    fn write_fmt(&mut self, args: fmt::Arguments<'_>) -> Result<()> {
1644        #[cfg(feature = "nightly")]
1645        {
1646            if let Some(s) = args.as_statically_known_str() {
1647                self.write_all(s.as_bytes())
1648            } else {
1649                default_write_fmt(self, args)
1650            }
1651        }
1652
1653        #[cfg(not(feature = "nightly"))]
1654        {
1655            default_write_fmt(self, args)
1656        }
1657    }

1659    /// Creates a "by reference" adapter for this instance of `Write`.
1660    ///
1661    /// The returned adapter also implements `Write` and will simply borrow this
1662    /// current writer.
1663    ///
1664    /// # Examples
1665    ///
1666    /// ```no_run
1667    /// use std::fs::File;
1668    ///
1669    /// use io_core::io::{self, Write};

1672    ///     let mut buffer = File::create("foo.txt")?;
1673    ///

1675    ///
1676    ///     // we can use reference just like our original buffer
1677    ///     reference.write_all(b"some bytes")?;
1678    ///     Ok(())
1679    /// }

1689/// The `Seek` trait provides a cursor which can be moved within a stream of
1690/// bytes.
1691///
1692/// The stream typically has a fixed size, allowing seeking relative to either
1693/// end or the current offset.
1694///
1695/// # Examples
1696///
1697/// [`File`]s implement `Seek`:
1698///
1699/// [`File`]: std::fs::File
1700///
1701/// ```no_run
1702/// use std::fs::File;
1703///
1704/// use io_core::io::{self, Seek, SeekFrom};
1705///
1706/// fn main() -> io::Result<()> {
1707///     let mut f = File::open("foo.txt")?;
1708///
1709///     // move the cursor 42 bytes from the start of the file
1710///     f.seek(SeekFrom::Start(42))?;
1711///     Ok(())
1712/// }
1713/// ```
1714pub trait Seek {
1715    /// Seek to an offset, in bytes, in a stream.
1716    ///
1717    /// A seek beyond the end of a stream is allowed, but behavior is defined
1718    /// by the implementation.
1719    ///
1720    /// If the seek operation completed successfully,
1721    /// this method returns the new position from the start of the stream.
1722    /// That position can be used later with [`SeekFrom::Start`].
1723    ///
1724    /// # Errors
1725    ///
1726    /// Seeking can fail, for example because it might involve flushing a buffer.
1727    ///
1728    /// Seeking to a negative offset is considered an error.
1729    fn seek(&mut self, pos: SeekFrom) -> Result<u64>;
1730
1731    /// Rewind to the beginning of a stream.
1732    ///
1733    /// This is a convenience method, equivalent to `seek(SeekFrom::Start(0))`.
1734    ///
1735    /// # Errors
1736    ///
1737    /// Rewinding can fail, for example because it might involve flushing a buffer.
1738    ///
1739    /// # Example
1740    ///
1741    /// ```no_run
1742    /// use std::fs::OpenOptions;
1743    ///
1744    /// use io_core::io::{self, Read, Seek, SeekFrom, Write};
1745    ///
1746    /// let mut f = OpenOptions::new().write(true).read(true).create(true).open("foo.txt")?;
1747    ///
1748    /// let hello = "Hello!\n";
1749    /// write!(f, "{hello}")?;
1750    /// f.rewind()?;
1751    ///
1752    /// let mut buf = String::new();
1753    /// f.read_to_string(&mut buf)?;
1754    /// assert_eq!(&buf, hello);
1755    /// # io_core::io::Result::Ok(())
1756    /// ```
1757    fn rewind(&mut self) -> Result<()> {
1758        self.seek(SeekFrom::Start(0))?;
1759        Ok(())
1760    }
1761
1762    /// Returns the length of this stream (in bytes).
1763    ///
1764    /// The default implementation uses up to three seek operations. If this
1765    /// method returns successfully, the seek position is unchanged (i.e. the
1766    /// position before calling this method is the same as afterwards).
1767    /// However, if this method returns an error, the seek position is
1768    /// unspecified.
1769    ///
1770    /// If you need to obtain the length of *many* streams and you don't care
1771    /// about the seek position afterwards, you can reduce the number of seek
1772    /// operations by simply calling `seek(SeekFrom::End(0))` and using its
1773    /// return value (it is also the stream length).
1774    ///
1775    /// Note that length of a stream can change over time (for example, when
1776    /// data is appended to a file). So calling this method multiple times does
1777    /// not necessarily return the same length each time.
1778    ///
1779    /// # Example
1780    ///
1781    /// ```no_run
1782    /// use std::fs::File;
1783    ///
1784    /// use io_core::io::{self, Seek};
1785    ///
1786    /// fn main() -> io::Result<()> {
1787    ///     let mut f = File::open("foo.txt")?;
1788    ///
1789    ///     let len = f.stream_len()?;
1790    ///     println!("The file is currently {len} bytes long");
1791    ///     Ok(())
1792    /// }
1793    /// ```
1794    fn stream_len(&mut self) -> Result<u64> {
1795        stream_len_default(self)
1796    }
1797
1798    /// Returns the current seek position from the start of the stream.
1799    ///
1800    /// This is equivalent to `self.seek(SeekFrom::Current(0))`.
1801    ///
1802    /// # Example
1803    ///
1804    /// ```no_run
1805    /// use std::fs::File;
1806    ///
1807    /// use io_core::io::{self, BufRead, BufReader, Seek, SeekFrom};
1808    ///
1809    /// fn main() -> io::Result<()> {
1810    ///     let mut f = BufReader::new(File::open("foo.txt")?);
1811    ///
1812    ///     let before = f.stream_position()?;
1813    ///     f.read_line(&mut String::new())?;
1814    ///     let after = f.stream_position()?;
1815    ///
1816    ///     println!("The first line was {} bytes long", after - before);
1817    ///     Ok(())
1818    /// }
1819    /// ```
1820    fn stream_position(&mut self) -> Result<u64> {
1821        self.seek(SeekFrom::Current(0))
1822    }
1823
1824    /// Seeks relative to the current position.
1825    ///
1826    /// This is equivalent to `self.seek(SeekFrom::Current(offset))` but
1827    /// doesn't return the new position which can allow some implementations
1828    /// such as [`BufReader`] to perform more efficient seeks.
1829    ///
1830    /// # Example
1831    ///
1832    /// ```no_run
1833    /// use std::fs::File;
1834    ///
1835    /// use io_core::io::{self, Seek};
1836    ///
1837    /// fn main() -> io::Result<()> {
1838    ///     let mut f = File::open("foo.txt")?;
1839    ///     f.seek_relative(10)?;
1840    ///     assert_eq!(f.stream_position()?, 10);
1841    ///     Ok(())
1842    /// }
1843    /// ```
1844    ///
1845    /// [`BufReader`]: crate::io::BufReader
1846    fn seek_relative(&mut self, offset: i64) -> Result<()> {
1847        self.seek(SeekFrom::Current(offset))?;
1848        Ok(())
1849    }
1850}

1852pub(crate) fn stream_len_default<T: Seek + ?Sized>(self_: &mut T) -> Result<u64> {
1853    let old_pos = self_.stream_position()?;
1854    let len = self_.seek(SeekFrom::End(0))?;
1855
1856    // Avoid seeking a third time when we were already at the end of the
1857    // stream. The branch is usually way cheaper than a seek operation.
1858    if old_pos != len {
1859        self_.seek(SeekFrom::Start(old_pos))?;
1860    }
1861
1862    Ok(len)
1863}

1888#[cfg(feature = "alloc")]
1889fn read_until<R: BufRead + ?Sized>(r: &mut R, delim: u8, buf: &mut Vec<u8>) -> Result<usize> {
1890    let mut read = 0;
1891    loop {
1892        let (done, used) = {
1893            let available = match r.fill_buf() {
1894                Ok(n) => n,
1895                Err(ref e) if e.is_interrupted() => continue,
1896                Err(e) => return Err(e),
1897            };
1898            match memchr::memchr(delim, available) {
1899                Some(i) => {
1900                    buf.extend_from_slice(&available[..=i]);
1901                    (true, i + 1)
1902                },
1903                None => {
1904                    buf.extend_from_slice(available);
1905                    (false, available.len())
1906                },
1907            }
1908        };
1909        r.consume(used);
1910        read += used;
1911        if done || used == 0 {
1912            return Ok(read);
1913        }
1914    }
1915}

1917fn skip_until<R: BufRead + ?Sized>(r: &mut R, delim: u8) -> Result<usize> {
1918    let mut read = 0;
1919    loop {
1920        let (done, used) = {
1921            let available = match r.fill_buf() {
1922                Ok(n) => n,
1923                Err(ref e) if e.kind() == ErrorKind::Interrupted => continue,
1924                Err(e) => return Err(e),
1925            };
1926            match memchr::memchr(delim, available) {
1927                Some(i) => (true, i + 1),
1928                None => (false, available.len()),
1929            }
1930        };
1931        r.consume(used);
1932        read += used;
1933        if done || used == 0 {
1934            return Ok(read);
1935        }
1936    }
1937}

1939/// A `BufRead` is a type of `Read`er which has an internal buffer, allowing it
1940/// to perform extra ways of reading.
1941///
1942/// For example, reading line-by-line is inefficient without using a buffer, so
1943/// if you want to read by line, you'll need `BufRead`, which includes a
1944/// [`read_line`] method as well as a [`lines`] iterator.
1945///
1946/// # Examples
1947///
1948/// A locked standard input implements `BufRead`:
1949///
1950/// ```no_run
1951/// use io_core::io::{BufRead, Read};
1952///
1953/// let stdin = std::io::stdin();
1954/// for line in stdin.lock().lines() {
1955///     println!("{}", line?);
1956/// }
1957/// # io_core::io::Result::Ok(())
1958/// ```
1959///
1960/// If you have something that implements [`Read`], you can use the [`BufReader`
1961/// type][`BufReader`] to turn it into a `BufRead`.
1962///
1963/// For example, [`File`] implements [`Read`], but not `BufRead`.
1964/// [`BufReader`] to the rescue!
1965///
1966/// [`File`]: std::fs::File
1967/// [`read_line`]: BufRead::read_line
1968/// [`lines`]: BufRead::lines
1969///
1970/// ```no_run
1971/// use std::fs::File;
1972///
1973/// use io_core::io::{self, BufRead, BufReader, Read};
1974///
1975/// fn main() -> io::Result<()> {
1976///     let f = File::open("foo.txt")?;
1977///     let f = BufReader::new(f);
1978///
1979///     for line in f.lines() {
1980///         let line = line?;
1981///         println!("{line}");
1982///     }
1983///
1984///     Ok(())
1985/// }
1986/// ```

1988    /// Returns the contents of the internal buffer, filling it with more data, via `Read` methods,
1989    /// if empty.
1990    ///
1991    /// This is a lower-level method and is meant to be used together with [`consume`],
1992    /// which can be used to mark bytes that should not be returned by subsequent calls to `read`.
1993    ///

1997    ///

2480                self.done_first = true;
2481            }
2482        }

2699impl<T: Read> Read for Take<T> {
2700    fn read(&mut self, buf: &mut [u8]) -> Result<usize> {
2701        // Don't call into inner reader at all at EOF because it may still block
2702        if self.limit == 0 {
2703            return Ok(0);
2704        }
2705
2706        let max = cmp::min(buf.len() as u64, self.limit) as usize;
2707        let n = self.inner.read(&mut buf[..max])?;
2708        assert!(n as u64 <= self.limit, "number of read bytes exceeds limit");
2709        self.limit -= n as u64;
2710        Ok(n)
2711    }
2712
2713    #[cfg(feature = "nightly")]
2714    fn read_buf(&mut self, mut buf: BorrowedCursor<'_, u8>) -> Result<()> {
2715        // Don't call into inner reader at all at EOF because it may still block
2716        if self.limit == 0 {
2717            return Ok(());
2718        }
2719
2720        if self.limit < buf.capacity() as u64 {
2721            // The condition above guarantees that `self.limit` fits in `usize`.
2722            let limit = self.limit as usize;
2723
2724            let is_init = buf.is_init();
2725
2726            // SAFETY: no uninit data is written to ibuf
2727            let ibuf = unsafe { &mut buf.as_mut()[..limit] };
2728
2729            let mut sliced_buf: BorrowedBuf<'_, u8> = ibuf.into();
2730
2731            if is_init {
2732                // SAFETY: `sliced_buf` is a subslice of `buf`, so if `buf` was initialized then
2733                // `sliced_buf` is.
2734                unsafe { sliced_buf.set_init() };
2735            }
2736
2737            let mut cursor = sliced_buf.unfilled();
2738            let result = self.inner.read_buf(cursor.reborrow());
2739
2740            let should_init = cursor.is_init();
2741            let filled = sliced_buf.len();
2742
2743            // cursor / sliced_buf / ibuf must drop here
2744
2745            // Avoid accidentally quadratic behaviour by initializing the whole
2746            // cursor if only part of it was initialized.
2747            if should_init {
2748                // SAFETY: no uninit data is written
2749                let uninit = unsafe { &mut buf.as_mut()[limit..] };
2750                uninit.write_filled(0);
2751                // SAFETY: all bytes that were not initialized by `T::read_buf`
2752                // have just been written to.
2753                unsafe { buf.set_init() };
2754            }
2755
2756            unsafe {
2757                // SAFETY: filled bytes have been filled
2758                buf.advance(filled);
2759            }
2760
2761            self.limit -= filled as u64;
2762
2763            result
2764        } else {
2765            let written = buf.written();
2766            let result = self.inner.read_buf(buf.reborrow());
2767            self.limit -= (buf.written() - written) as u64;
2768            result
2769        }
2770    }
2771}

2774    fn fill_buf(&mut self) -> Result<&[u8]> {
2775        // Don't call into inner reader at all at EOF because it may still block
2776        if self.limit == 0 {
2777            return Ok(&[]);
2778        }
2779
2780        let buf = self.inner.fill_buf()?;
2781        let cap = cmp::min(buf.len() as u64, self.limit) as usize;
2782        Ok(&buf[..cap])
2783    }

2785    fn consume(&mut self, amt: usize) {
2786        // Don't let callers reset the limit by passing an overlarge value
2787        let amt = cmp::min(amt as u64, self.limit) as usize;
2788        self.limit -= amt as u64;
2789        self.inner.consume(amt);
2790    }

2809impl<T: Seek> Seek for Take<T> {
2810    fn seek(&mut self, pos: SeekFrom) -> Result<u64> {
2811        let new_position = match pos {
2812            SeekFrom::Start(v) => Some(v),
2813            SeekFrom::Current(v) => self.position().checked_add_signed(v),
2814            SeekFrom::End(v) => self.len.checked_add_signed(v),
2815        };
2816        let new_position = match new_position {
2817            Some(v) if v <= self.len => v,
2818            _ => return Err(ErrorKind::InvalidInput.into()),
2819        };
2820        while new_position != self.position() {
2821            if let Some(offset) = new_position.checked_signed_diff(self.position()) {
2822                self.inner.seek_relative(offset)?;
2823                self.limit = self.limit.wrapping_sub(offset as u64);
2824                break;
2825            }
2826            let offset = if new_position > self.position() { i64::MAX } else { i64::MIN };
2827            self.inner.seek_relative(offset)?;
2828            self.limit = self.limit.wrapping_sub(offset as u64);
2829        }
2830        Ok(new_position)
2831    }

2839    }

2841    fn seek_relative(&mut self, offset: i64) -> Result<()> {
2842        if self.position().checked_add_signed(offset).is_none_or(|p| p > self.len) {
2843            return Err(ErrorKind::InvalidInput.into());
2844        }
2845        self.inner.seek_relative(offset)?;
2846        self.limit = self.limit.wrapping_sub(offset as u64);
2847        Ok(())
2848    }
