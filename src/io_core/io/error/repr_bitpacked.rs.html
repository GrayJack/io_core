1//! This is a densely packed error representation which is used on targets with
2//! 64-bit pointers.
3//!
4//! (Note that `bitpacked` vs `unpacked` here has no relationship to
5//! `#[repr(packed)]`, it just refers to attempting to use any available bits in
6//! a more clever manner than `rustc`'s default layout algorithm would).
7//!
8//! Conceptually, it stores the same data as the "unpacked" equivalent we use on
9//! other targets. Specifically, you can imagine it as an optimized version of
10//! the following enum (which is roughly equivalent to what's stored by
11//! `repr_unpacked::Repr`, e.g. `super::ErrorData<Box<Custom>>`):
12//!
13//! ```ignore (exposition-only)
14//! enum ErrorData {
15//!    Os(i32),
16//!    Simple(ErrorKind),
17//!    SimpleMessage(&'static SimpleMessage),
18//!    Custom(Box<Custom>),
19//! }
20//! ```
21//!
22//! However, it packs this data into a 64bit non-zero value.
23//!
24//! This optimization not only allows `io::Error` to occupy a single pointer,
25//! but improves `io::Result` as well, especially for situations like
26//! `io::Result<()>` (which is now 64 bits) or `io::Result<u64>` (which is now
27//! 128 bits), which are quite common.
28//!
29//! # Layout
30//! Tagged values are 64 bits, with the 2 least significant bits used for the
31//! tag. This means there are 4 "variants":
32//!
33//! - **Tag 0b00**: The first variant is equivalent to `ErrorData::SimpleMessage`, and holds a
34//!   `&'static SimpleMessage` directly.
35//!
36//!   `SimpleMessage` has an alignment >= 4 (which is requested with
37//!   `#[repr(align)]` and checked statically at the bottom of this file), which
38//!   means every `&'static SimpleMessage` should have the both tag bits as 0,
39//!   meaning its tagged and untagged representation are equivalent.
40//!
41//!   This means we can skip tagging it, which is necessary as this variant can
42//!   be constructed from a `const fn`, which probably cannot tag pointers (or
43//!   at least it would be difficult).
44//!
45//! - **Tag 0b01**: The other pointer variant holds the data for `ErrorData::Custom` and the
46//!   remaining 62 bits are used to store a `Box<Custom>`. `Custom` also has alignment >= 4, so the
47//!   bottom two bits are free to use for the tag.
48//!
49//!   The only important thing to note is that `ptr::wrapping_add` and
50//!   `ptr::wrapping_sub` are used to tag the pointer, rather than bitwise
51//!   operations. This should preserve the pointer's provenance, which would
52//!   otherwise be lost.
53//!
54//! - **Tag 0b10**: Holds the data for `ErrorData::Os(i32)`. We store the `i32` in the pointer's
55//!   most significant 32 bits, and don't use the bits `2..32` for anything. Using the top 32 bits
56//!   is just to let us easily recover the `i32` code with the correct sign.
57//!
58//! - **Tag 0b11**: Holds the data for `ErrorData::Simple(ErrorKind)`. This stores the `ErrorKind`
59//!   in the top 32 bits as well, although it doesn't occupy nearly that many. Most of the bits are
60//!   unused here, but it's not like we need them for anything else yet.
61//!
62//! # Use of `NonNull<()>`
63//!
64//! Everything is stored in a `NonNull<()>`, which is odd, but actually serves a
65//! purpose.
66//!
67//! Conceptually you might think of this more like:
68//!
69//! ```ignore (exposition-only)
70//! union Repr {
71//!     // holds integer (Simple/Os) variants, and
72//!     // provides access to the tag bits.
73//!     bits: NonZero<u64>,
74//!     // Tag is 0, so this is stored untagged.
75//!     msg: &'static SimpleMessage,
76//!     // Tagged (offset) `Box<Custom>` pointer.
77//!     tagged_custom: NonNull<()>,
78//! }
79//! ```
80//!
81//! But there are a few problems with this:
82//!
83//! 1. Union access is equivalent to a transmute, so this representation would require we transmute
84//!    between integers and pointers in at least one direction, which may be UB (and even if not, it
85//!    is likely harder for a compiler to reason about than explicit ptr->int operations).
86//!
87//! 2. Even if all fields of a union have a niche, the union itself doesn't, although this may
88//!    change in the future. This would make things like `io::Result<()>` and `io::Result<usize>`
89//!    larger, which defeats part of the motivation of this bitpacking.
90//!
91//! Storing everything in a `NonZero<usize>` (or some other integer) would be a
92//! bit more traditional for pointer tagging, but it would lose provenance
93//! information, couldn't be constructed from a `const fn`, and would probably
94//! run into other issues as well.
95//!
96//! The `NonNull<()>` seems like the only alternative, even if it's fairly odd
97//! to use a pointer type to store something that may hold an integer, some of
98//! the time.
99

107// The 2 least-significant bits are used as tag.
108const TAG_MASK: usize = 0b11;
109const TAG_SIMPLE_MESSAGE: usize = 0b00;
110const TAG_CUSTOM: usize = 0b01;
111const TAG_OS: usize = 0b10;
112const TAG_SIMPLE: usize = 0b11;

114/// The internal representation.
115///
116/// See the module docs for more, this is just a way to hack in a check that we
117/// indeed are not unwind-safe.
118///
119/// ```compile_fail,E0277
120/// fn is_unwind_safe<T: core::panic::UnwindSafe>() {}
121/// is_unwind_safe::<std::io::Error>();
122/// ```
123#[repr(transparent)]
124pub(super) struct Repr(NonNull<()>, PhantomData<ErrorData<CustomOwner>>);
125
126// All the types `Repr` stores internally are Send + Sync, and so is it.
127unsafe impl Send for Repr {}
128unsafe impl Sync for Repr {}
129
130impl Repr {
131    pub(super) fn new_custom(b: CustomOwner) -> Self {
132        let p = b.into_raw().as_ptr().cast::<u8>();
133        // Should only be possible if an allocator handed out a pointer with
134        // wrong alignment.
135        debug_assert_eq!(p.addr() & TAG_MASK, 0);
136        // Note: We know `TAG_CUSTOM <= size_of::<Custom>()` (static_assert at
137        // end of file), and both the start and end of the expression must be
138        // valid without address space wraparound due to `Box`'s semantics.
139        //
140        // This means it would be correct to implement this using `ptr::add`
141        // (rather than `ptr::wrapping_add`), but it's unclear this would give
142        // any benefit, so we just use `wrapping_add` instead.
143        let tagged = p.wrapping_add(TAG_CUSTOM).cast::<()>();
144        // SAFETY: `TAG_CUSTOM + p` is the same as `TAG_CUSTOM | p`,
145        // because `p`'s alignment means it isn't allowed to have any of the
146        // `TAG_BITS` set (you can verify that addition and bitwise-or are the
147        // same when the operands have no bits in common using a truth table).
148        //
149        // Then, `TAG_CUSTOM | p` is not zero, as that would require
150        // `TAG_CUSTOM` and `p` both be zero, and neither is (as `p` came from a
151        // box, and `TAG_CUSTOM` just... isn't zero -- it's `0b01`). Therefore,
152        // `TAG_CUSTOM + p` isn't zero and so `tagged` can't be, and the
153        // `new_unchecked` is safe.
154        let ptr = unsafe { NonNull::new_unchecked(tagged) };
155        let res = Self(ptr, PhantomData);
156        // quickly smoke-check we encoded the right thing (This generally will
157        // only run in std's tests, unless the user uses -Zbuild-std)
158        debug_assert!(matches!(res.data(), ErrorData::Custom(_)), "repr(custom) encoding failed");
159        res
160    }
161
162    #[inline]
163    pub(super) fn new_os(code: RawOsError) -> Self {
164        let utagged = ((code as usize) << 32) | TAG_OS;
165        // SAFETY: `TAG_OS` is not zero, so the result of the `|` is not 0.
166        let utagged = unsafe { NonZeroUsize::new_unchecked(utagged) };
167        let res = Self(NonNull::without_provenance(utagged), PhantomData);
168        // quickly smoke-check we encoded the right thing (This generally will
169        // only run in std's tests, unless the user uses -Zbuild-std)
170        debug_assert!(
171            matches!(res.data(), ErrorData::Os(c) if c == code),
172            "repr(os) encoding failed for {code}"
173        );
174        res
175    }
176
177    #[inline]
178    pub(super) fn new_simple(kind: ErrorKind) -> Self {
179        let utagged = ((kind as usize) << 32) | TAG_SIMPLE;
180        // SAFETY: `TAG_SIMPLE` is not zero, so the result of the `|` is not 0.
181        let utagged = unsafe { NonZeroUsize::new_unchecked(utagged) };
182        let res = Self(NonNull::without_provenance(utagged), PhantomData);
183        // quickly smoke-check we encoded the right thing (This generally will
184        // only run in std's tests, unless the user uses -Zbuild-std)
185        debug_assert!(
186            matches!(res.data(), ErrorData::Simple(k) if k == kind),
187            "repr(simple) encoding failed {:?}",
188            kind,
189        );
190        res
191    }
192
193    #[inline]
194    pub(super) const fn new_simple_message(m: &'static SimpleMessage) -> Self {
195        // SAFETY: References are never null.
196        let ptr = unsafe { NonNull::new_unchecked(m as *const _ as *mut ()) };
197        Self(ptr, PhantomData)
198    }
199
200    #[inline]
201    pub(super) fn data(&self) -> ErrorData<&Custom> {
202        // SAFETY: We're a Repr, decode_repr is fine.
203        unsafe { decode_repr(self.0, |c| &*c) }
204    }
205
206    #[inline]
207    pub(super) fn data_mut(&mut self) -> ErrorData<&mut Custom> {
208        // SAFETY: We're a Repr, decode_repr is fine.
209        unsafe { decode_repr(self.0, |c| &mut *c) }
210    }
211
212    #[inline]
213    #[cfg(feature = "alloc")]
214    pub(super) fn into_data(self) -> ErrorData<CustomOwner> {
215        let this = core::mem::ManuallyDrop::new(self);
216        // SAFETY: We're a Repr, decode_repr is fine. The `Box::from_raw` is
217        // safe because we prevent double-drop using `ManuallyDrop`.
218        unsafe { decode_repr(this.0, |p| CustomOwner::from_raw(NonNull::new_unchecked(p))) }
219    }
220}
221
222impl Drop for Repr {
223    #[inline]
224    fn drop(&mut self) {
225        // SAFETY: We're a Repr, decode_repr is fine. The `Box::from_raw` is
226        // safe because we're being dropped.
227        unsafe {
228            let _ = decode_repr(self.0, |p| CustomOwner::from_raw(NonNull::new_unchecked(p)));
229        }
230    }
231}

233// Shared helper to decode a `Repr`'s internal pointer into an ErrorData.
234//
235// SAFETY: `ptr`'s bits should be encoded as described in the document at the
236// top (it should `some_repr.0`)
237#[inline]
238unsafe fn decode_repr<C, F>(ptr: NonNull<()>, make_custom: F) -> ErrorData<C>
239where
240    F: FnOnce(*mut Custom) -> C,
241{
242    let bits = ptr.as_ptr().addr();
243    match bits & TAG_MASK {
244        TAG_OS => {
245            let code = ((bits as i64) >> 32) as RawOsError;
246            ErrorData::Os(code)
247        },
248        TAG_SIMPLE => {
249            let kind_bits = (bits >> 32) as u32;
250            let kind = kind_from_prim(kind_bits).unwrap_or_else(|| {
251                debug_assert!(false, "Invalid io::error::Repr bits: `Repr({:#018x})`", bits);
252                // This means the `ptr` passed in was not valid, which violates
253                // the unsafe contract of `decode_repr`.
254                //
255                // Using this rather than unwrap meaningfully improves the code
256                // for callers which only care about one variant (usually
257                // `Custom`)
258                unsafe { core::hint::unreachable_unchecked() };
259            });
260            ErrorData::Simple(kind)
261        },
262        TAG_SIMPLE_MESSAGE => {
263            // SAFETY: per tag
264            unsafe { ErrorData::SimpleMessage(&*ptr.cast::<SimpleMessage>().as_ptr()) }
265        },
266        TAG_CUSTOM => {
267            // It would be correct for us to use `ptr::byte_sub` here (see the
268            // comment above the `wrapping_add` call in `new_custom` for why),
269            // but it isn't clear that it makes a difference, so we don't.
270            let custom = ptr.as_ptr().wrapping_byte_sub(TAG_CUSTOM).cast::<Custom>();
271            ErrorData::Custom(make_custom(custom))
272        },
273        _ => {
274            // Can't happen, and compiler can tell
275            unreachable!();
276        },
277    }
278}

280// This compiles to the same code as the check+transmute, but doesn't require
281// unsafe, or to hard-code max ErrorKind or its size in a way the compiler
282// couldn't verify.
283#[inline]
284fn kind_from_prim(ek: u32) -> Option<ErrorKind> {
285    macro_rules! from_prim {
286        ($prim:expr => $Enum:ident { $($Variant:ident),* $(,)? }) => {{
287            // Force a compile error if the list gets out of date.
288            const _: fn(e: $Enum) = |e: $Enum| match e {
289                $($Enum::$Variant => ()),*
290            };
291            match $prim {
292                $(v if v == ($Enum::$Variant as _) => Some($Enum::$Variant),)*
293                _ => None,
294            }
295        }}
296    }
297    from_prim!(ek => ErrorKind {
298        NotFound,
299        PermissionDenied,
300        ConnectionRefused,
301        ConnectionReset,
302        HostUnreachable,
303        NetworkUnreachable,
304        ConnectionAborted,
305        NotConnected,
306        AddrInUse,
307        AddrNotAvailable,
308        NetworkDown,
309        BrokenPipe,
310        AlreadyExists,
311        WouldBlock,
312        NotADirectory,
313        IsADirectory,
314        DirectoryNotEmpty,
315        ReadOnlyFilesystem,
316        FilesystemLoop,
317        StaleNetworkFileHandle,
318        InvalidInput,
319        InvalidData,
320        TimedOut,
321        WriteZero,
322        StorageFull,
323        NotSeekable,
324        QuotaExceeded,
325        FileTooLarge,
326        ResourceBusy,
327        ExecutableFileBusy,
328        Deadlock,
329        CrossesDevices,
330        TooManyLinks,
331        InvalidFilename,
332        ArgumentListTooLong,
333        Interrupted,
334        Other,
335        UnexpectedEof,
336        Unsupported,
337        OutOfMemory,
338        InProgress,
339        Uncategorized,
340    })
341}

343// Some static checking to alert us if a change breaks any of the assumptions
344// that our encoding relies on for correctness and soundness. (Some of these are
345// a bit overly thorough/cautious, admittedly)
346//
347// If any of these are hit on a platform that std supports, we should likely
348// just use `repr_unpacked.rs` there instead (unless the fix is easy).
349macro_rules! static_assert {
350    ($condition:expr) => {
351        const _: () = assert!($condition);
352    };
353    (@usize_eq: $lhs:expr, $rhs:expr) => {
354        const _: [(); $lhs] = [(); $rhs];
355    };
356}
357
358// The bitpacking we use requires pointers be exactly 64 bits.
359static_assert!(@usize_eq: size_of::<NonNull<()>>(), 8);
360
361// We also require pointers and usize be the same size.
362static_assert!(@usize_eq: size_of::<NonNull<()>>(), size_of::<usize>());
363
364// `Custom` and `SimpleMessage` need to be thin pointers.
365static_assert!(@usize_eq: size_of::<&'static SimpleMessage>(), 8);
366static_assert!(@usize_eq: size_of::<CustomOwner>(), 8);
367
368static_assert!((TAG_MASK + 1).is_power_of_two());
369// And they must have sufficient alignment.
370static_assert!(align_of::<SimpleMessage>() > TAG_MASK);
371static_assert!(align_of::<Custom>() > TAG_MASK);
372
373static_assert!(@usize_eq: TAG_MASK & TAG_SIMPLE_MESSAGE, TAG_SIMPLE_MESSAGE);
374static_assert!(@usize_eq: TAG_MASK & TAG_CUSTOM, TAG_CUSTOM);
375static_assert!(@usize_eq: TAG_MASK & TAG_OS, TAG_OS);
376static_assert!(@usize_eq: TAG_MASK & TAG_SIMPLE, TAG_SIMPLE);
377
378// This is obviously true (`TAG_CUSTOM` is `0b01`), but in `Repr::new_custom` we
379// offset a pointer by this value, and expect it to both be within the same
380// object, and to not wrap around the address space. See the comment in that
381// function for further details.
382//
383// Actually, at the moment we use `ptr::wrapping_add`, not `ptr::add`, so this
384// check isn't needed for that one, although the assertion that we don't
385// actually wrap around in that wrapping_add does simplify the safety reasoning
386// elsewhere considerably.
387static_assert!(size_of::<Custom>() >= TAG_CUSTOM);
388
389// These two store a payload which is allowed to be zero, so they must be
390// non-zero to preserve the `NonNull`'s range invariant.
391static_assert!(TAG_OS != 0);
392static_assert!(TAG_SIMPLE != 0);
393// We can't tag `SimpleMessage`s, the tag must be 0.
394static_assert!(@usize_eq: TAG_SIMPLE_MESSAGE, 0);
395
396// Check that the point of all of this still holds.
397//
398// We'd check against `io::Error`, but *technically* it's allowed to vary,
399// as it's not `#[repr(transparent)]`/`#[repr(C)]`. We could add that, but
400// the `#[repr()]` would show up in rustdoc, which might be seen as a stable
401// commitment.
402static_assert!(@usize_eq: size_of::<Repr>(), 8);
403static_assert!(@usize_eq: size_of::<Option<Repr>>(), 8);
404static_assert!(@usize_eq: size_of::<Result<(), Repr>>(), 8);
405static_assert!(@usize_eq: size_of::<Result<usize, Repr>>(), 16);
