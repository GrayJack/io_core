12/// A borrowed byte buffer which is incrementally filled.
13///
14/// This type makes it safer to work with `MaybeUninit` buffers, such as to read into a buffer
15/// without having to initialize it first. It tracks the region of bytes that have been filled and
16/// whether the unfilled region was initialized.
17///
18/// In summary, the contents of the buffer can be visualized as:
19/// ```not_rust
20/// [                capacity                ]
21/// [ filled | unfilled (may be initialized) ]
22/// ```
23///
24/// A `BorrowedBuf` is created around some existing data (or capacity for data) via a unique
25/// reference (`&mut`). The `BorrowedBuf` can be configured (e.g., using `clear` or `set_init`), but
26/// cannot be directly written. To write into the buffer, use `unfilled` to create a
27/// `BorrowedCursor`. The cursor has write-only access to the unfilled portion of the buffer (you
28/// can think of it as a write-only iterator).
29///
30/// The lifetime `'data` is a bound on the lifetime of the underlying data.
31pub struct BorrowedBuf<'data, T> {
32    /// The buffer's underlying data.
33    buf: &'data mut [MaybeUninit<T>],
34    /// The length of `self.buf` which is known to be filled.
35    filled: usize,
36    /// Whether the entire unfilled part of `self.buf` has explicitly been initialized.
37    init: bool,
38}

119/// Creates a new `BorrowedBuf` from a cursor.
120///
121/// Use `BorrowedCursor::with_unfilled_buf` instead for a safer alternative.
122impl<'data, T: Copy> From<BorrowedCursor<'data, T>> for BorrowedBuf<'data, T> {
123    #[inline]
124    fn from(buf: BorrowedCursor<'data, T>) -> BorrowedBuf<'data, T> {
125        let filled = buf.filled();
126        let init = buf.is_buf_init();
127        let len = buf.buf_len();
128        BorrowedBuf {
129            // SAFETY: no initialized element is ever uninitialized as per `BorrowedBuf`'s
130            // invariant, and the cursor holds the unique access to those elements for `'data`
131            buf: unsafe { slice::from_raw_parts_mut(buf.buf.as_ptr().add(filled), len - filled) },
132            filled: 0,
133            init,
134        }
135    }
136}

159impl<'data, T: Copy> BorrowedBuf<'data, T> {
160    /// Returns a shared reference to the filled portion of the buffer.
161    #[inline]
162    pub fn filled(&self) -> &[T] {
163        // SAFETY: We only slice the filled part of the buffer, which is always valid
164        unsafe {
165            let buf = self.buf.get_unchecked(..self.filled);
166            buf.assume_init_ref()
167        }
168    }
169
170    /// Returns a mutable reference to the filled portion of the buffer.
171    #[inline]
172    pub fn filled_mut(&mut self) -> &mut [T] {
173        // SAFETY: We only slice the filled part of the buffer, which is always valid
174        unsafe {
175            let buf = self.buf.get_unchecked_mut(..self.filled);
176            buf.assume_init_mut()
177        }
178    }
179
180    /// Returns a shared reference to the filled portion of the buffer with its original lifetime.
181    #[inline]
182    pub fn into_filled(self) -> &'data [T] {
183        // SAFETY: We only slice the filled part of the buffer, which is always valid
184        unsafe {
185            let buf = self.buf.get_unchecked(..self.filled);
186            buf.assume_init_ref()
187        }
188    }
189
190    /// Returns a mutable reference to the filled portion of the buffer with its original lifetime.
191    #[inline]
192    pub fn into_filled_mut(self) -> &'data mut [T] {
193        // SAFETY: We only slice the filled part of the buffer, which is always valid
194        unsafe {
195            let buf = self.buf.get_unchecked_mut(..self.filled);
196            buf.assume_init_mut()
197        }
198    }
199
200    /// Returns a cursor over the unfilled part of the buffer.
201    #[inline]
202    pub const fn unfilled<'this>(&'this mut self) -> BorrowedCursor<'this, T> {
203        let borrowed_buf = NonNull::from_mut(self);
204        BorrowedCursor {
205            buf: NonNull::from_mut(self.buf).cast(),
206            borrowed_buf,
207        }
208    }
209
210    /// Clears the buffer, resetting the filled region to empty.
211    ///
212    /// The contents of the buffer are not modified.
213    #[inline]
214    pub const fn clear(&mut self) -> &mut Self {
215        self.filled = 0;
216        self
217    }
218
219    /// Asserts that the unfilled part of the buffer is initialized.
220    ///
221    /// # Safety
222    ///
223    /// All the bytes of the buffer must be initialized.
224    #[inline]
225    pub const unsafe fn set_init(&mut self) -> &mut Self {
226        self.init = true;
227        self
228    }

231/// A writeable view of the unfilled portion of a [`BorrowedBuf`].
232///
233/// The unfilled portion may be uninitialized; see [`BorrowedBuf`] for details.
234///
235/// Data can be written directly to the cursor by using [`append`](BorrowedCursor::append) or
236/// indirectly by getting a slice of part or all of the cursor and writing into the slice. In the
237/// indirect case, the caller must call [`advance`](BorrowedCursor::advance) after writing to inform
238/// the cursor how many bytes have been written.
239///
240/// Once data is written to the cursor, it becomes part of the filled portion of the underlying
241/// `BorrowedBuf` and can no longer be accessed or re-written by the cursor. I.e., the cursor tracks
242/// the unfilled part of the underlying `BorrowedBuf`.
243///
244/// The lifetime `'a` is a bound on the lifetime of the underlying buffer (which means it is a bound
245/// on the data in that buffer by transitivity).
246pub struct BorrowedCursor<'a, T> {
247    /// The start of the elements of the buffer this cursor was created from.
248    /// Safety invariant: this points to the start of the *whole* buffer of `*borrowed_buf` and is
249    /// valid for reads and writes of `(*borrowed_buf).buf.len()` elements, so that
250    /// `(*borrowed_buf).filled` indexes into it.
251    buf: NonNull<MaybeUninit<T>>,
252    /// The buffer this cursor was created from.
253    /// Safety invariants:
254    /// 1. `(*borrowed_buf).buf` is *never* accessed by the owner of the pointee while the `buf`
255    ///    field above is alive, because there is a `&mut` of the pointee while the cursor is
256    ///    alive.
257    /// 2. We promise to only access the `filled` and `init` fields and the metadata of the `buf`
258    ///    field through the `borrowed_buf` pointer, never triggering any retag of `buf`'s pointer,
259    ///    as the `buf` field above holds a reborrow of it and reaching the parent again would be a
260    ///    foreign access for that reborrow. This includes not making a reference to the whole
261    ///    pointee out of `borrowed_buf`, but only accessing those fields directly through pointer
262    ///    manipulation.
263    borrowed_buf: NonNull<BorrowedBuf<'a, T>>,
264}
265
266impl<T> Debug for BorrowedCursor<'_, T> {
267    fn fmt(&self, f: &mut Formatter<'_>) -> fmt::Result {
268        let buf = BorrowedBufDebug {
269            init: self.is_buf_init(),
270            filled: self.filled(),
271            capacity: self.buf_len(),
272        };
273
274        f.debug_struct("BorrowedCursor").field("buf", &buf).finish()
275    }

278// Helpers to access underlying buffer state.
279impl<'a, T> BorrowedCursor<'a, T> {
280    #[inline]
281    const fn buf_mut(&mut self) -> &mut [MaybeUninit<T>] {
282        let len = self.buf_len();
283        // SAFETY: `buf` points to `len` elements that this cursor borrows exclusively.
284        unsafe { slice::from_raw_parts_mut(self.buf.as_ptr(), len) }
285    }
286
287    #[inline]
288    const fn buf_len(&self) -> usize {
289        // SAFETY: We read just the metadata of `buf` and avoid retagging the reference.
290        unsafe {
291            let borrowed_buf = self.borrowed_buf.as_ptr();
292            let buf_ptr: *const &'a mut [MaybeUninit<T>] = &raw const (*borrowed_buf).buf;
293            // Same layout:
294            // https://doc.rust-lang.org/reference/type-layout.html#r-layout.pointer.intro
295            let buf_ptr: *const *const [MaybeUninit<T>] = buf_ptr.cast();
296            let buf: *const [MaybeUninit<T>] = *buf_ptr;
297            buf.len()
298        }
299    }
300
301    #[inline]
302    fn unfilled_slice(&mut self) -> &mut [MaybeUninit<T>] {
303        let filled = self.filled();
304        // SAFETY: always in bounds
305        unsafe { self.buf_mut().get_unchecked_mut(filled..) }
306    }
307
308    #[inline]
309    const fn filled(&self) -> usize {
310        // SAFETY: We access just `filled` and avoid foreign read on `buf`.
311        unsafe { (*self.borrowed_buf.as_ptr()).filled }
312    }
313
314    #[inline]
315    const fn is_buf_init(&self) -> bool {
316        // SAFETY: We access just `init` and avoid foreign read on `buf`.
317        unsafe { (*self.borrowed_buf.as_ptr()).init }
318    }
319
320    /// # Safety
321    ///
322    /// In case of `true` all the elements of the cursor must be initialized.
323    #[inline]
324    const unsafe fn set_buf_init(&mut self, init: bool) {
325        // SAFETY: We access just `init` and avoid foreign read on `buf`.
326        unsafe {
327            (*self.borrowed_buf.as_ptr()).init = init;
328        }
329    }
330
331    /// # Safety
332    ///
333    /// The next `n` elements of the cursor must be initialized.
334    #[inline]
335    const unsafe fn add_filled(&mut self, n: usize) {
336        // SAFETY: We access just `filled` and avoid foreign read on `buf`.
337        unsafe {
338            (*self.borrowed_buf.as_ptr()).filled += n;
339        }
340    }
341}

343impl<'a, T: Copy> BorrowedCursor<'a, T> {
344    /// Reborrows this cursor by cloning it with a smaller lifetime.
345    ///
346    /// Since a cursor maintains unique access to its underlying buffer, the borrowed cursor is
347    /// not accessible while the new cursor exists.
348    #[inline]
349    pub const fn reborrow<'this>(&'this mut self) -> BorrowedCursor<'this, T> {
350        BorrowedCursor {
351            buf: self.buf,
352            borrowed_buf: self.borrowed_buf,
353        }
354    }
355
356    /// Returns the available space in the cursor.
357    #[inline]
358    pub const fn capacity(&self) -> usize {
359        self.buf_len() - self.filled()
360    }
361
362    /// Returns the number of bytes written to the `BorrowedBuf` this cursor was created from.
363    ///
364    /// In particular, the count returned is shared by all reborrows of the cursor.
365    #[inline]
366    pub const fn written(&self) -> usize {
367        self.filled()
368    }
369
370    /// Returns `true` if the buffer is initialized.
371    #[inline]
372    pub const fn is_init(&self) -> bool {
373        self.is_buf_init()
374    }
375
376    /// Set the buffer as fully initialized.
377    ///
378    /// # Safety
379    ///
380    /// All the bytes of the cursor must be initialized.
381    #[inline]
382    pub const unsafe fn set_init(&mut self) {
383        // SAFETY: the caller guarantees that all the elements of the cursor are initialized.
384        unsafe { self.set_buf_init(true) }
385    }
386
387    /// Returns a mutable reference to the whole cursor.
388    ///
389    /// # Safety
390    ///
391    /// The caller must not uninitialize any bytes of the cursor if it is initialized.
392    #[inline]
393    pub unsafe fn as_mut(&mut self) -> &mut [MaybeUninit<T>] {
394        self.unfilled_slice()
395    }
396
397    /// Advances the cursor by asserting that `n` bytes have been filled.
398    ///
399    /// After advancing, the `n` bytes are no longer accessible via the cursor and can only be
400    /// accessed via the underlying buffer. I.e., the buffer's filled portion grows by `n` elements
401    /// and its unfilled portion (and the capacity of this cursor) shrinks by `n` elements.
402    ///
403    /// If less than `n` bytes initialized (by the cursor's point of view), `set_init` should be
404    /// called first.
405    ///
406    /// # Panics
407    ///
408    /// Panics if there are less than `n` bytes initialized.
409    #[inline]
410    pub const fn advance_checked(&mut self, n: usize) -> &mut Self {
411        // The subtraction cannot underflow by invariant of this type.
412        let init_unfilled = if self.is_buf_init() {
413            self.buf_len() - self.filled()
414        } else {
415            0
416        };
417        assert!(n <= init_unfilled);
418
419        // SAFETY: the next `n` elements are initialized, as asserted above.
420        unsafe { self.advance(n) };
421        self
422    }
423
424    /// Advances the cursor by asserting that `n` bytes have been filled.
425    ///
426    /// After advancing, the `n` bytes are no longer accessible via the cursor and can only be
427    /// accessed via the underlying buffer. I.e., the buffer's filled portion grows by `n` elements
428    /// and its unfilled portion (and the capacity of this cursor) shrinks by `n` elements.
429    ///
430    /// # Safety
431    ///
432    /// The caller must ensure that the first `n` bytes of the cursor have been properly
433    /// initialised.
434    #[inline]
435    pub const unsafe fn advance(&mut self, n: usize) -> &mut Self {
436        // SAFETY: the caller guarantees that the first `n` elements of the cursor are initialized.
437        unsafe { self.add_filled(n) };
438        self
439    }
440
441    /// Appends data to the cursor, advancing position within its buffer.
442    ///
443    /// # Panics
444    ///
445    /// Panics if `self.capacity()` is less than `buf.len()`.
446    #[inline]
447    pub fn append(&mut self, buf: &[T]) {
448        assert!(self.capacity() >= buf.len());
449
450        // SAFETY: we do not de-initialize any of the elements of the slice
451        unsafe {
452            self.as_mut()[..buf.len()].write_copy_of_slice(buf);
453        }
454
455        // SAFETY: these elements have just been initialized.
456        unsafe { self.advance(buf.len()) };
457    }
458
459    /// Runs the given closure with a `BorrowedBuf` containing the unfilled part
460    /// of the cursor.
461    ///
462    /// This enables inspecting what was written to the cursor.
463    ///
464    /// # Panics
465    ///
466    /// Panics if the `BorrowedBuf` given to the closure is replaced by another
467    /// one.
468    pub fn with_unfilled_buf<R>(&mut self, f: impl FnOnce(&mut BorrowedBuf<'_, T>) -> R) -> R {
469        let mut buf = BorrowedBuf::from(self.reborrow());
470        let prev_ptr = buf.buf as *const _;
471        let res = f(&mut buf);
472
473        // Check that the caller didn't replace the `BorrowedBuf`.
474        // This is necessary for the safety of the code below: if the check wasn't
475        // there, one could mark some elements as initialized even though they aren't.
476        assert!(core::ptr::eq(prev_ptr, buf.buf));
477
478        let filled = buf.filled;
479        let init = buf.init;
480
481        // Update `init` and `filled` fields with what was written to the buffer.
482        // `self.buf.filled` was the starting length of the `BorrowedBuf`.
483        //
484        // SAFETY: These elements were initialized/filled in the `BorrowedBuf`, and therefore they
485        // are initialized/filled in the cursor too, because the buffer wasn't replaced.
486        unsafe {
487            self.set_buf_init(init);
488            self.advance(filled);
489        }
490
491        res
492    }
493}

495impl<'a, T: Default + Copy> BorrowedCursor<'a, T> {
496    /// Initializes all elements in the cursor with their default value and
497    /// returns them.
498    #[inline]
499    pub fn ensure_init(&mut self) -> &mut [T] {
500        if !self.is_init() {
501            cfg_select! {
502                all(feature = "nightly", feature = "__write_default") => {
503                    self.unfilled_slice().write_default()
504                },
505                all(feature = "nightly", not(feature = "__write_default")) => {
506                    maybeuninit_write_default(self.unfilled_slice())
507                },
508                _ => {
509                    self.unfilled_slice().fill(MaybeUninit::new(Default::default()));
510                },
511            };
512
513            unsafe { self.set_buf_init(true) };
514        }
515
516        // SAFETY: these elements have just been initialized if they weren't before
517        unsafe { self.unfilled_slice().assume_init_mut() }
518    }
519}

521/// A borrowed byte buffer which is incrementally filled and initialized.
522///
523/// This type is a sort of "double cursor". It tracks three regions in the buffer: a region at the
524/// beginning of the buffer that has been logically filled with data, a region that has been
525/// initialized at some point but not yet logically filled, and a region at the end that is fully
526/// uninitialized. The filled region is guaranteed to be a subset of the initialized region.
527///
528/// In summary, the contents of the buffer can be visualized as:
529/// ```not_rust
530/// [             capacity              ]
531/// [ filled |         unfilled         ]
532/// [    initialized    | uninitialized ]
533/// ```
534///
535/// A `TrackingBorrowedBuf` is created around some existing data (or capacity for data) via a unique
536/// reference (`&mut`). The `TrackingBorrowedBuf` can be configured (e.g., using `clear` or
537/// `set_init`), but cannot be directly written. To write into the buffer, use `unfilled` to create
538/// a `TrackingBorrowedCursor`. The cursor has write-only access to the unfilled portion of the
539/// buffer (you can think of it as a write-only iterator).
540///
541/// The lifetime `'data` is a bound on the lifetime of the underlying data.
542pub struct TrackingBorrowedBuf<'data, T> {
543    /// The buffer's underlying data.
544    buf: &'data mut [MaybeUninit<T>],
545    /// The length of `self.buf` which is known to be filled.
546    filled: usize,
547    /// The length of `self.buf` which is known to be initialized.
548    init: usize,
549}
550
551impl<T> fmt::Debug for TrackingBorrowedBuf<'_, T> {
552    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
553        TrackingBorrowedBufDebug {
554            init: self.init,
555            filled: self.filled,
556            capacity: self.capacity(),
557        }
558        .fmt(f)
559    }
560}

639/// Use `TrackingBorrowedCursor::with_unfilled_buf` instead for a safer alternative.
640impl<'data, T: Copy> From<TrackingBorrowedCursor<'data, T>> for TrackingBorrowedBuf<'data, T> {
641    #[inline]
642    fn from(buf: TrackingBorrowedCursor<'data, T>) -> TrackingBorrowedBuf<'data, T> {
643        let filled = buf.filled();
644        let init = buf.init();
645        let len = buf.buf_len();
646        TrackingBorrowedBuf {
647            // SAFETY: no initialized element is ever uninitialized as per `BorrowedBuf`'s
648            // invariant, and the cursor holds the unique access to those elements for `'data`
649            buf: unsafe { slice::from_raw_parts_mut(buf.buf.as_ptr().add(filled), len - filled) },
650            filled: 0,
651            init,
652        }
653    }
654}

682impl<'data, T: Copy> TrackingBorrowedBuf<'data, T> {
683    /// Returns a shared reference to the filled portion of the buffer.
684    #[inline]
685    pub fn filled(&self) -> &[T] {
686        // SAFETY: We only slice the filled part of the buffer, which is always valid
687        unsafe {
688            let buf = self.buf.get_unchecked(..self.filled);
689            buf.assume_init_ref()
690        }
691    }
692
693    /// Returns a mutable reference to the filled portion of the buffer.
694    #[inline]
695    pub fn filled_mut(&mut self) -> &mut [T] {
696        // SAFETY: We only slice the filled part of the buffer, which is always valid
697        unsafe {
698            let buf = self.buf.get_unchecked_mut(..self.filled);
699            buf.assume_init_mut()
700        }
701    }
702
703    /// Returns a shared reference to the filled portion of the buffer with its original lifetime.
704    #[inline]
705    pub fn into_filled(self) -> &'data [T] {
706        // SAFETY: We only slice the filled part of the buffer, which is always valid
707        unsafe {
708            let buf = self.buf.get_unchecked(..self.filled);
709            buf.assume_init_ref()
710        }
711    }
712
713    /// Returns a mutable reference to the filled portion of the buffer with its original lifetime.
714    #[inline]
715    pub fn into_filled_mut(self) -> &'data mut [T] {
716        // SAFETY: We only slice the filled part of the buffer, which is always valid
717        unsafe {
718            let buf = self.buf.get_unchecked_mut(..self.filled);
719            buf.assume_init_mut()
720        }
721    }
722
723    /// Returns a cursor over the unfilled part of the buffer.
724    #[inline]
725    pub const fn unfilled<'this>(&'this mut self) -> TrackingBorrowedCursor<'this, T> {
726        let borrowed_buf = NonNull::from_mut(self);
727        TrackingBorrowedCursor {
728            buf: NonNull::from_mut(self.buf).cast(),
729            borrowed_buf,
730            start: self.filled,
731        }
732    }
733
734    /// Clears the buffer, resetting the filled region to empty.
735    ///
736    /// The number of initialized bytes is not changed, and the contents of the buffer are not
737    /// modified.
738    #[inline]
739    pub const fn clear(&mut self) -> &mut Self {
740        self.filled = 0;
741        self
742    }
743
744    /// Asserts that the first `n` bytes of the buffer are initialized.
745    ///
746    /// `BorrowedBuf` assumes that bytes are never de-initialized, so this method does nothing when
747    /// called with fewer bytes than are already known to be initialized.
748    ///
749    /// # Safety
750    ///
751    /// The caller must ensure that the first `n` unfilled bytes of the buffer have already been
752    /// initialized.
753    #[inline]
754    pub const unsafe fn set_init(&mut self, n: usize) -> &mut Self {
755        self.init = max(self.init, n);
756        self
757    }
758}

760/// A writeable view of the unfilled portion of a [`TrackingBorrowedBuf`].
761///
762/// Provides access to the initialized and uninitialized parts of the underlying
763/// `TrackingBorrowedBuf`. Data can be written directly to the cursor by using
764/// [`append`](TrackingBorrowedCursor::append) or indirectly by getting a slice of part or all of
765/// the cursor and writing into the slice. In the indirect case, the caller must call
766/// [`advance`](TrackingBorrowedCursor::advance) after writing to inform the cursor how many bytes
767/// have been written.
768///
769/// Once data is written to the cursor, it becomes part of the filled portion of the underlying
770/// `TrackingBorrowedBuf` and can no longer be accessed or re-written by the cursor. I.e., the
771/// cursor tracks the unfilled part of the underlying `TrackingBorrowedBuf`.
772///
773/// The lifetime `'a` is a bound on the lifetime of the underlying buffer (which means it is a bound
774/// on the data in that buffer by transitivity).
775pub struct TrackingBorrowedCursor<'a, T> {
776    /// The start of the elements of the buffer this cursor was created from.
777    /// Safety invariant: this points to the start of the *whole* buffer of `*borrowed_buf` and is
778    /// valid for reads and writes of `(*borrowed_buf).buf.len()` elements, so that
779    /// `(*borrowed_buf).filled` indexes into it.
780    buf: NonNull<MaybeUninit<T>>,
781    /// The buffer this cursor was created from.
782    /// Safety invariants:
783    /// 1. `(*borrowed_buf).buf` is *never* accessed by the owner of the pointee while the `buf`
784    ///    field above is alive, because there is a `&mut` of the pointee while the cursor is
785    ///    alive.
786    /// 2. We promise to only access the `filled` and `init` fields and the metadata of the `buf`
787    ///    field through the `borrowed_buf` pointer, never triggering any retag of `buf`'s pointer,
788    ///    as the `buf` field above holds a reborrow of it and reaching the parent again would be a
789    ///    foreign access for that reborrow. This includes not making a reference to the whole
790    ///    pointee out of `borrowed_buf`, but only accessing those fields directly through pointer
791    ///    manipulation.
792    borrowed_buf: NonNull<TrackingBorrowedBuf<'a, T>>,
793    /// The length of the filled portion of the underlying buffer at the time of the cursor's
794    /// creation.
795    start: usize,
796}

798impl<T> Debug for TrackingBorrowedCursor<'_, T> {
799    fn fmt(&self, f: &mut Formatter<'_>) -> fmt::Result {
800        let buf = TrackingBorrowedBufDebug {
801            init: self.init(),
802            filled: self.filled(),
803            capacity: self.buf_len(),
804        };
805
806        f.debug_struct("BorrowedCursor").field("buf", &buf).finish()
807    }
808}

810// Helpers to access underlying buffer state.
811impl<'a, T> TrackingBorrowedCursor<'a, T> {
812    #[inline]
813    const fn buf(&self) -> &[MaybeUninit<T>] {
814        let len = self.buf_len();
815        // SAFETY: `buf` points to `len` elements that this cursor borrows exclusively.
816        unsafe { slice::from_raw_parts(self.buf.as_ptr().cast_const(), len) }
817    }
818
819    #[inline]
820    const fn buf_mut(&mut self) -> &mut [MaybeUninit<T>] {
821        let len = self.buf_len();
822        // SAFETY: `buf` points to `len` elements that this cursor borrows exclusively.
823        unsafe { slice::from_raw_parts_mut(self.buf.as_ptr(), len) }
824    }
825
826    #[inline]
827    const fn buf_len(&self) -> usize {
828        // SAFETY: We read just the metadata of `buf` and avoid retagging the reference.
829        unsafe {
830            let borrowed_buf = self.borrowed_buf.as_ptr();
831            let buf_ptr: *const &'a mut [MaybeUninit<T>] = &raw const (*borrowed_buf).buf;
832            // Same layout:
833            // https://doc.rust-lang.org/reference/type-layout.html#r-layout.pointer.intro
834            let buf_ptr: *const *const [MaybeUninit<T>] = buf_ptr.cast();
835            let buf: *const [MaybeUninit<T>] = *buf_ptr;
836            buf.len()
837        }
838    }
839
840    #[inline]
841    fn unfilled_slice(&mut self) -> &mut [MaybeUninit<T>] {
842        let filled = self.filled();
843        // SAFETY: always in bounds
844        unsafe { self.buf_mut().get_unchecked_mut(filled..) }
845    }
846
847    #[inline]
848    fn init_slice_ref(&self) -> &[MaybeUninit<T>] {
849        let filled = self.filled();
850        let init = self.init();
851        // SAFETY: always in bounds
852        unsafe { self.buf().get_unchecked(filled..init) }
853    }
854
855    #[inline]
856    fn init_slice(&mut self) -> &mut [MaybeUninit<T>] {
857        let filled = self.filled();
858        let init = self.init();
859        // SAFETY: always in bounds
860        unsafe { self.buf_mut().get_unchecked_mut(filled..init) }
861    }
862
863    #[inline]
864    fn uninit_slice(&mut self) -> &mut [MaybeUninit<T>] {
865        let init = self.init();
866        // SAFETY: always in bounds
867        unsafe { self.buf_mut().get_unchecked_mut(init..) }
868    }
869
870    #[inline]
871    const fn filled(&self) -> usize {
872        // SAFETY: We access just `filled` and avoid foreign read on `buf`.
873        unsafe { (*self.borrowed_buf.as_ptr()).filled }
874    }
875
876    #[inline]
877    const fn init(&self) -> usize {
878        // SAFETY: We access just `init` and avoid foreign read on `buf`.
879        unsafe { (*self.borrowed_buf.as_ptr()).init }
880    }
881
882    /// # Safety
883    ///
884    /// In case of `true` all the elements of the cursor must be initialized.
885    #[inline]
886    const unsafe fn set_buf_init(&mut self, init: usize) {
887        // SAFETY: We access just `init` and avoid foreign read on `buf`.
888        unsafe {
889            (*self.borrowed_buf.as_ptr()).init = init;
890        }
891    }
892
893    /// # Safety
894    ///
895    /// The next `n` elements of the cursor must be initialized.
896    #[inline]
897    const unsafe fn add_filled(&mut self, n: usize) {
898        // SAFETY: We access just `filled` and avoid foreign read on `buf`.
899        unsafe {
900            (*self.borrowed_buf.as_ptr()).filled += n;
901        }
902    }
903}

905impl<'a, T: Copy> TrackingBorrowedCursor<'a, T> {
906    /// Reborrows this cursor by cloning it with a smaller lifetime.
907    ///
908    /// Since a cursor maintains unique access to its underlying buffer, the borrowed cursor is
909    /// not accessible while the new cursor exists.
910    #[inline]
911    pub const fn reborrow<'this>(&'this mut self) -> TrackingBorrowedCursor<'this, T> {
912        TrackingBorrowedCursor {
913            buf: self.buf,
914            borrowed_buf: self.borrowed_buf,
915            start: self.start,
916        }
917    }
918
919    /// Returns the available space in the cursor.
920    #[inline]
921    pub const fn capacity(&self) -> usize {
922        self.buf_len() - self.filled()
923    }
924
925    /// Returns the number of bytes written to this cursor since it was created from a
926    /// `BorrowedBuf`.
927    ///
928    /// Note that if this cursor is a reborrowed clone of another, then the count returned is the
929    /// count written via either cursor, not the count since the cursor was reborrowed.
930    #[inline]
931    pub const fn written(&self) -> usize {
932        self.filled() - self.start
933    }
934
935    /// Returns a shared reference to the initialized portion of the cursor.
936    #[inline]
937    pub fn init_ref(&self) -> &[T] {
938        let buf = self.init_slice_ref();
939        // SAFETY: We only slice the initialized part of the buffer, which is always valid
940        unsafe { buf.assume_init_ref() }
941    }
942
943    /// Returns a mutable reference to the initialized portion of the cursor.
944    #[inline]
945    pub fn init_mut(&mut self) -> &mut [T] {
946        let buf = self.init_slice();
947        // SAFETY: We only slice the initialized part of the buffer, which is always valid
948        unsafe { buf.assume_init_mut() }
949    }
950
951    /// Returns a mutable reference to the uninitialized part of the cursor.
952    ///
953    /// It is safe to uninitialize any of these bytes.
954    #[inline]
955    pub fn uninit_mut(&mut self) -> &mut [MaybeUninit<T>] {
956        self.uninit_slice()
957    }
958
959    /// Returns a mutable reference to the whole cursor.
960    ///
961    /// # Safety
962    ///
963    /// The caller must not uninitialize any bytes in the initialized portion of the cursor.
964    #[inline]
965    pub unsafe fn as_mut(&mut self) -> &mut [MaybeUninit<T>] {
966        self.unfilled_slice()
967    }
968
969    /// Advance the cursor by asserting that `n` bytes have been filled.
970    ///
971    /// After advancing, the `n` bytes are no longer accessible via the cursor and can only be
972    /// accessed via the underlying buffer. I.e., the buffer's filled portion grows by `n` elements
973    /// and its unfilled portion (and the capacity of this cursor) shrinks by `n` elements.
974    ///
975    /// If less than `n` bytes initialized (by the cursor's point of view), `set_init` should be
976    /// called first.
977    ///
978    /// # Panics
979    ///
980    /// Panics if there are less than `n` bytes initialized.
981    #[inline]
982    pub const fn advance_checked(&mut self, n: usize) -> &mut Self {
983        let filled = self.filled().strict_add(n);
984        let init = self.init();
985        assert!(filled <= init);
986
987        // SAFETY: the next `n` elements are initialized, as asserted above.
988        unsafe { self.advance(n) };
989        self
990    }
991
992    /// Advance the cursor by asserting that `n` bytes have been filled.
993    ///
994    /// After advancing, the `n` bytes are no longer accessible via the cursor and can only be
995    /// accessed via the underlying buffer. I.e., the buffer's filled portion grows by `n` elements
996    /// and its unfilled portion (and the capacity of this cursor) shrinks by `n` elements.
997    ///
998    /// # Safety
999    ///
1000    /// The caller must ensure that the first `n` bytes of the cursor have been properly
1001    /// initialized.
1002    #[inline]
1003    pub const unsafe fn advance(&mut self, n: usize) -> &mut Self {
1004        // SAFETY: the caller guarantees that the first `n` elements of the cursor are initialized.
1005        unsafe {
1006            self.add_filled(n);
1007            self.set_buf_init(max(self.init(), self.filled()));
1008        }
1009        self
1010    }
1011
1012    /// Asserts that the first `n` unfilled bytes of the cursor are initialized.
1013    ///
1014    /// `BorrowedBuf` assumes that bytes are never de-initialized, so this method does nothing when
1015    /// called with fewer bytes than are already known to be initialized.
1016    ///
1017    /// # Safety
1018    ///
1019    /// The caller must ensure that the first `n` bytes of the buffer have already been initialized.
1020    #[inline]
1021    pub unsafe fn set_init(&mut self, n: usize) -> &mut Self {
1022        unsafe { self.set_buf_init(cmp::max(self.init(), self.filled() + n)) };
1023        self
1024    }
1025
1026    /// Appends data to the cursor, advancing position within its buffer.
1027    ///
1028    /// # Panics
1029    ///
1030    /// Panics if `self.capacity()` is less than `buf.len()`.
1031    #[inline]
1032    pub fn append(&mut self, buf: &[T]) {
1033        assert!(self.capacity() >= buf.len());
1034
1035        // SAFETY: we do not de-initialize any of the elements of the slice
1036        unsafe { self.as_mut()[..buf.len()].write_copy_of_slice(buf) };
1037
1038        // SAFETY: We just added the entire contents of buf to the filled section.
1039        unsafe {
1040            self.set_init(buf.len());
1041            self.advance(buf.len());
1042        }
1043    }
1044
1045    /// Runs the given closure with a `BorrowedBuf` containing the unfilled part
1046    /// of the cursor.
1047    ///
1048    /// This enables inspecting what was written to the cursor.
1049    ///
1050    /// # Panics
1051    ///
1052    /// Panics if the `BorrowedBuf` given to the closure is replaced by another
1053    /// one.
1054    pub fn with_unfilled_buf<R>(
1055        &mut self, f: impl FnOnce(&mut TrackingBorrowedBuf<'_, T>) -> R,
1056    ) -> R {
1057        let mut buf = TrackingBorrowedBuf::from(self.reborrow());
1058        let prev_ptr = buf.buf as *const _;
1059        let res = f(&mut buf);
1060
1061        // Check that the caller didn't replace the `BorrowedBuf`.
1062        // This is necessary for the safety of the code below: if the check wasn't
1063        // there, one could mark some elements as initialized even though they aren't.
1064        assert!(core::ptr::eq(prev_ptr, buf.buf));
1065
1066        let filled = buf.filled;
1067        let init = buf.init;
1068
1069        // Update `init` and `filled` fields with what was written to the buffer.
1070        // `self.buf.filled` was the starting length of the `BorrowedBuf`.
1071        //
1072        // SAFETY: These elements were initialized/filled in the `BorrowedBuf`, and therefore they
1073        // are initialized/filled in the cursor too, because the buffer wasn't replaced.
1074        unsafe {
1075            self.set_buf_init(init);
1076            self.advance(filled);
1077        }
1078
1079        res
1080    }
1081}

1083impl<'a, T: Default + Copy> TrackingBorrowedCursor<'a, T> {
1084    /// Initializes all elements in the cursor with their default value and
1085    /// returns them.
1086    ///
1087    /// This includes items not filled and not initialized.
1088    #[inline]
1089    pub fn ensure_init(&'a mut self) -> &'a mut [T] {
1090        let capacity = self.capacity();
1091
1092        if self.init() > 0 {
1093            // SAFETY: always in bounds and we never uninitialize these elements.
1094            let unfilled = self.unfilled_slice();
1095
1096            cfg_select! {
1097                all(feature = "nightly", feature = "__write_default") => unfilled.write_default(),
1098                all(feature = "nightly", not(feature = "__write_default")) => {
1099                    maybeuninit_write_default(unfilled)
1100                },
1101                _ => {
1102                    unfilled.fill(MaybeUninit::new(Default::default()));
1103                },
1104            };
1105        }
1106
1107        // SAFETY: these elements have just been initialized if they weren't before
1108        unsafe {
1109            self.set_buf_init(capacity);
1110            self.unfilled_slice().assume_init_mut()
1111        }
1112    }
1113}

1115#[cfg(feature = "nightly")]
1116fn maybeuninit_write_default<T>(buf: &mut [MaybeUninit<T>]) -> &mut [T]
1117where
1118    T: Default,
1119{
1120    trait DefaultSpec: Default {
1121        fn write_default(buf: &mut [MaybeUninit<Self>]) -> &mut [Self];
1122    }
1123
1124    impl<T: Default> DefaultSpec for T {
1125        default fn write_default(buf: &mut [MaybeUninit<Self>]) -> &mut [Self] {
1126            buf.write_with(|_| T::default())
1127        }
1128    }
1129
1130    macro_rules! spec_default_zero {
1131        ($ty:ty) => {
1132            impl DefaultSpec for $ty {
1133                fn write_default(buf: &mut [MaybeUninit<Self>]) -> &mut [Self] {
1134                    // SAFETY:
1135                    // `Default::default` is equivalent to zero-initialization
1136                    // for all these types, and this initializes the entire
1137                    // slice.
1138                    unsafe {
1139                        buf.as_mut_ptr().write_bytes(0, buf.len());
1140                        buf.assume_init_mut()
1141                    }
1142                }
1143            }
1144        };
1145    }
1146
1147    spec_default_zero!(i8);
1148    spec_default_zero!(u8);
1149    spec_default_zero!(i16);
1150    spec_default_zero!(u16);
1151    spec_default_zero!(i32);
1152    spec_default_zero!(u32);
1153    spec_default_zero!(i64);
1154    spec_default_zero!(u64);
1155    spec_default_zero!(i128);
1156    spec_default_zero!(u128);
1157    spec_default_zero!(isize);
1158    spec_default_zero!(usize);
1159
1160    T::write_default(buf)
1161}

1166    } else {
